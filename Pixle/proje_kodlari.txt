using System.Collections.Generic;
using System.Numerics;
using Raylib_cs;

namespace PixelArtApp.Core;

public class Canvas
{
    public int Width { get; }
    public int Height { get; }
    public int PixelScale { get; }
    public Vector2 Offset { get; set; }
    public bool ShowGrid { get; set; } = true;
    public static readonly Color TransparentColor = new Color(0, 0, 0, 0);

    // YENÄ°: Animasyon Kareleri (Frames)
    public List<Frame> Frames { get; private set; } = new();
    public int ActiveFrameIndex { get; set; } = 0;

    // Eskiden var olan tÃ¼m sistemlerin bozulmadan Ã§alÄ±ÅŸmasÄ± iÃ§in yÃ¶nlendiriciler
    public List<Layer> Layers => Frames[ActiveFrameIndex].Layers;
    public int ActiveLayerIndex 
    { 
        get => Frames[ActiveFrameIndex].ActiveLayerIndex; 
        set => Frames[ActiveFrameIndex].ActiveLayerIndex = value; 
    }

    public Canvas(int width, int height, int scale, Vector2 offset)
    {
        Width = width;
        Height = height;
        PixelScale = scale;
        Offset = offset;
        
        // Ä°lk aÃ§Ä±lÄ±ÅŸta 1. Kare (Frame) ve iÃ§ine 1. KatmanÄ± (Layer) ekle
        Frames.Add(new Frame());
        Frames[0].Layers.Add(new Layer("Katman 1", width, height));
    }

    public void AddLayer()
    {
        Layers.Add(new Layer($"Katman {Layers.Count + 1}", Width, Height));
        ActiveLayerIndex = Layers.Count - 1; 
    }

    // YENÄ°: Animasyon Karesi Ekleme (Bir Ã¶nceki kareyi kopyalarak devam eder)
    public void AddFrame()
    {
        var newFrame = new Frame();
        foreach (var l in Layers)
        {
            newFrame.Layers.Add(l.Clone()); // Son Ã§izimi kopyala
        }
        newFrame.ActiveLayerIndex = ActiveLayerIndex;
        Frames.Add(newFrame);
        ActiveFrameIndex = Frames.Count - 1;
    }

    public Color GetPixel(int x, int y)
    {
        if (x < 0 || x >= Width || y < 0 || y >= Height) return TransparentColor;
        return Layers[ActiveLayerIndex].Pixels[x, y];
    }

    public void SetPixel(int x, int y, Color color)
    {
        if (x < 0 || x >= Width || y < 0 || y >= Height) return;
        Layers[ActiveLayerIndex].Pixels[x, y] = color;
    }

    public Color GetMergedPixel(int x, int y)
    {
        if (x < 0 || x >= Width || y < 0 || y >= Height) return TransparentColor;
        Color final = TransparentColor;
        
        for (int i = 0; i < Layers.Count; i++)
        {
            if (!Layers[i].IsVisible) continue;
            Color c = Layers[i].Pixels[x, y];
            if (c.A > 0) final = c; 
        }
        return final;
    }

    public void Clear(Color color)
    {
        var p = Layers[ActiveLayerIndex].Pixels;
        for (int x = 0; x < Width; x++)
            for (int y = 0; y < Height; y++)
                p[x, y] = color;
    }

    public bool IsMouseInside(Vector2 worldPos)
    {
        return worldPos.X >= Offset.X && worldPos.X < Offset.X + Width * PixelScale &&
               worldPos.Y >= Offset.Y && worldPos.Y < Offset.Y + Height * PixelScale;
    }

    public (int x, int y) ScreenToGrid(Vector2 worldPos)
    {
        int gx = (int)((worldPos.X - Offset.X) / PixelScale);
        int gy = (int)((worldPos.Y - Offset.Y) / PixelScale);
        return (gx, gy);
    }

    public void Draw()
    {
        for (int x = 0; x < Width; x++)
        {
            for (int y = 0; y < Height; y++)
            {
                Color bg = ((x + y) % 2 == 0) ? new Color(200, 200, 200, 255) : new Color(160, 160, 160, 255);
                Raylib.DrawRectangle((int)Offset.X + x * PixelScale, (int)Offset.Y + y * PixelScale, PixelScale, PixelScale, bg);
            }
        }

        for (int i = 0; i < Layers.Count; i++)
        {
            if (!Layers[i].IsVisible) continue;
            var p = Layers[i].Pixels;
            for (int x = 0; x < Width; x++)
                for (int y = 0; y < Height; y++)
                    if (p[x, y].A > 0)
                        Raylib.DrawRectangle((int)Offset.X + x * PixelScale, (int)Offset.Y + y * PixelScale, PixelScale, PixelScale, p[x, y]);
        }

        if (ShowGrid)
        {
            for (int x = 0; x <= Width; x++)
                Raylib.DrawLine((int)Offset.X + x * PixelScale, (int)Offset.Y, (int)Offset.X + x * PixelScale, (int)Offset.Y + Height * PixelScale, new Color(0, 0, 0, 50));
            for (int y = 0; y <= Height; y++)
                Raylib.DrawLine((int)Offset.X, (int)Offset.Y + y * PixelScale, (int)Offset.X + Width * PixelScale, (int)Offset.Y + y * PixelScale, new Color(0, 0, 0, 50));
        }
    }
}

using System;
using System.Collections.Generic;

namespace PixelArtApp.Core;

public class HistoryManager
{
    private Stack<(List<Frame> frames, int activeFrame)> undoStack = new();
    private Stack<(List<Frame> frames, int activeFrame)> redoStack = new();

    public void SaveState(Canvas canvas)
    {
        var snapshot = new List<Frame>();
        foreach (var f in canvas.Frames) snapshot.Add(f.Clone());

        undoStack.Push((snapshot, canvas.ActiveFrameIndex));
        redoStack.Clear(); 
    }

    public void Undo(Canvas canvas)
    {
        if (undoStack.Count == 0) return;
        
        var currentSnapshot = new List<Frame>();
        foreach (var f in canvas.Frames) currentSnapshot.Add(f.Clone());
        redoStack.Push((currentSnapshot, canvas.ActiveFrameIndex));

        var previous = undoStack.Pop();
        canvas.Frames.Clear();
        foreach (var f in previous.frames) canvas.Frames.Add(f.Clone());
        canvas.ActiveFrameIndex = previous.activeFrame;
    }

    public void Redo(Canvas canvas)
    {
        if (redoStack.Count == 0) return;

        var currentSnapshot = new List<Frame>();
        foreach (var f in canvas.Frames) currentSnapshot.Add(f.Clone());
        undoStack.Push((currentSnapshot, canvas.ActiveFrameIndex));

        var next = redoStack.Pop();
        canvas.Frames.Clear();
        foreach (var f in next.frames) canvas.Frames.Add(f.Clone());
        canvas.ActiveFrameIndex = next.activeFrame;
    }
}

using System;
using Raylib_cs;

namespace PixelArtApp.Core;

public class Layer
{
    public string Name { get; set; }
    public Color[,] Pixels { get; set; }
    public bool IsVisible { get; set; } = true;

    public Layer(string name, int width, int height)
    {
        Name = name;
        Pixels = new Color[width, height];
        
        // KatmanÄ± tamamen ÅŸeffaf baÅŸlat
        for (int x = 0; x < width; x++)
        {
            for (int y = 0; y < height; y++)
            {
                Pixels[x, y] = new Color(0, 0, 0, 0); 
            }
        }
    }

    public Layer Clone()
    {
        int w = Pixels.GetLength(0);
        int h = Pixels.GetLength(1);
        Layer clone = new Layer(Name, w, h);
        clone.IsVisible = IsVisible;
        Array.Copy(Pixels, clone.Pixels, Pixels.Length);
        return clone;
    }
}

using System;
using System.Collections.Generic;
using System.Numerics;
using Raylib_cs;

namespace PixelArtApp.Core;

public enum SelectToolMode
{
    Rectangle,
    Lasso,
    MagicWand
}

public class SelectionManager
{
    public bool HasSelection { get; private set; } = false;
    public bool IsFloating { get; private set; } = false;
    public bool IsCreatingSelection { get; private set; } = false;
    public bool IsDraggingFloating { get; private set; } = false;

    private bool[,] mask = new bool[1, 1];
    private Color[,] floatingPixels = new Color[1, 1];

    public Rectangle Bounds { get; private set; }
    public Vector2 Offset { get; private set; }

    private (int x, int y) createStartPos;
    private (int x, int y) createCurrentPos;
    public List<(int x, int y)> LassoPath { get; private set; } = new();

    private (int x, int y) dragStartGrid;
    private Vector2 initialOffset;

    private Color[,]? clipboardPixels;
    private bool[,]? clipboardMask;
    private int clipWidth, clipHeight;

    public void EnsureSize(int width, int height)
    {
        if (mask.GetLength(0) != width || mask.GetLength(1) != height)
        {
            mask = new bool[width, height];
            floatingPixels = new Color[width, height];
        }
    }

    public bool IsPointInCurrentSelection(int gx, int gy)
    {
        if (!HasSelection) return false;
        int srcX = gx - (int)Offset.X;
        int srcY = gy - (int)Offset.Y;
        if (srcX < 0 || srcX >= mask.GetLength(0) || srcY < 0 || srcY >= mask.GetLength(1)) return false;
        return mask[srcX, srcY];
    }

    public Rectangle GetCurrentBounds()
    {
        return new Rectangle(Bounds.X + Offset.X, Bounds.Y + Offset.Y, Bounds.Width, Bounds.Height);
    }

    public void StartSelection(Canvas canvas, int gx, int gy, SelectToolMode mode, bool isAdd = false, bool isSubtract = false)
    {
        EnsureSize(canvas.Width, canvas.Height);

        // EÄŸer yÃ¼zen pikseller taÅŸÄ±nÄ±yorsa Ã¶nce onlarÄ± tuvale yapÄ±ÅŸtÄ±r
        if (IsFloating)
        {
            CancelOrCommit(canvas);
        }

        if (mode == SelectToolMode.MagicWand)
        {
            DoMagicWand(canvas, gx, gy, isAdd, isSubtract);
        }
        else
        {
            if (!isAdd && !isSubtract)
            {
                CancelOrCommit(canvas);
            }

            IsCreatingSelection = true;
            createStartPos = (gx, gy);
            createCurrentPos = (gx, gy);
            LassoPath.Clear();
            LassoPath.Add((gx, gy));
        }
    }

    public void UpdateSelectionCreation(int gx, int gy, SelectToolMode mode)
    {
        if (!IsCreatingSelection) return;
        createCurrentPos = (gx, gy);

        if (mode == SelectToolMode.Lasso)
        {
            var last = LassoPath[^1];
            if (last.x != gx || last.y != gy)
            {
                var line = ToolManager.GetLinePixels(last.x, last.y, gx, gy);
                foreach (var pt in line)
                {
                    if (LassoPath[^1] != pt)
                        LassoPath.Add(pt);
                }
            }
        }
    }

    public void EndSelectionCreation(Canvas canvas, SelectToolMode mode, bool isAdd = false, bool isSubtract = false)
    {
        if (!IsCreatingSelection) return;
        IsCreatingSelection = false;

        int w = canvas.Width;
        int h = canvas.Height;
        bool[,] tempMask = new bool[w, h];

        if (mode == SelectToolMode.Rectangle)
        {
            int minX = Math.Clamp(Math.Min(createStartPos.x, createCurrentPos.x), 0, w - 1);
            int maxX = Math.Clamp(Math.Max(createStartPos.x, createCurrentPos.x), 0, w - 1);
            int minY = Math.Clamp(Math.Min(createStartPos.y, createCurrentPos.y), 0, h - 1);
            int maxY = Math.Clamp(Math.Max(createStartPos.y, createCurrentPos.y), 0, h - 1);

            for (int x = minX; x <= maxX; x++)
            {
                for (int y = minY; y <= maxY; y++)
                {
                    tempMask[x, y] = true;
                }
            }
        }
        else if (mode == SelectToolMode.Lasso)
        {
            if (LassoPath.Count < 3)
            {
                int cx = Math.Clamp(createStartPos.x, 0, w - 1);
                int cy = Math.Clamp(createStartPos.y, 0, h - 1);
                tempMask[cx, cy] = true;
            }
            else
            {
                var closingLine = ToolManager.GetLinePixels(LassoPath[^1].x, LassoPath[^1].y, LassoPath[0].x, LassoPath[0].y);
                foreach (var pt in closingLine) LassoPath.Add(pt);

                for (int x = 0; x < w; x++)
                {
                    for (int y = 0; y < h; y++)
                    {
                        if (IsPointInPolygon(x, y, LassoPath))
                        {
                            tempMask[x, y] = true;
                        }
                    }
                }
                foreach (var (px, py) in LassoPath)
                {
                    if (px >= 0 && px < w && py >= 0 && py < h)
                        tempMask[px, py] = true;
                }
            }
        }

        // Maske BirleÅŸtirme (Ekleme / Ã‡Ä±karma / Yeni)
        for (int x = 0; x < w; x++)
        {
            for (int y = 0; y < h; y++)
            {
                if (isAdd)
                {
                    if (tempMask[x, y]) mask[x, y] = true;
                }
                else if (isSubtract)
                {
                    if (tempMask[x, y]) mask[x, y] = false;
                }
                else
                {
                    mask[x, y] = tempMask[x, y];
                }
            }
        }

        Offset = Vector2.Zero;
        RecalculateBounds(w, h);
    }

    private void DoMagicWand(Canvas canvas, int startX, int startY, bool isAdd, bool isSubtract)
    {
        int w = canvas.Width;
        int h = canvas.Height;
        if (startX < 0 || startX >= w || startY < 0 || startY >= h) return;

        Color target = canvas.GetPixel(startX, startY);

        if (target.A == 0)
        {
            if (!isAdd && !isSubtract)
            {
                CancelOrCommit(canvas);
            }
            return;
        }

        bool[,] newSelection = new bool[w, h];
        Queue<(int x, int y)> q = new();
        q.Enqueue((startX, startY));
        bool[,] visited = new bool[w, h];
        visited[startX, startY] = true;

        ReadOnlySpan<(int dx, int dy)> dirs = stackalloc (int, int)[] { (1, 0), (-1, 0), (0, 1), (0, -1) };

        while (q.Count > 0)
        {
            var (x, y) = q.Dequeue();
            newSelection[x, y] = true;

            foreach (var (dx, dy) in dirs)
            {
                int nx = x + dx;
                int ny = y + dy;
                if (nx >= 0 && nx < w && ny >= 0 && ny < h && !visited[nx, ny])
                {
                    visited[nx, ny] = true;
                    Color c = canvas.GetPixel(nx, ny);
                    if (c.R == target.R && c.G == target.G && c.B == target.B && c.A == target.A)
                    {
                        q.Enqueue((nx, ny));
                    }
                }
            }
        }

        if (!isAdd && !isSubtract)
        {
            Array.Clear(mask, 0, mask.Length);
        }

        for (int x = 0; x < w; x++)
        {
            for (int y = 0; y < h; y++)
            {
                if (newSelection[x, y])
                {
                    if (isSubtract)
                    {
                        mask[x, y] = false;
                    }
                    else
                    {
                        mask[x, y] = true;
                    }
                }
            }
        }

        Offset = Vector2.Zero;
        RecalculateBounds(w, h);
    }

    private bool IsPointInPolygon(int x, int y, List<(int x, int y)> poly)
    {
        bool inside = false;
        for (int i = 0, j = poly.Count - 1; i < poly.Count; j = i++)
        {
            if (((poly[i].y > y) != (poly[j].y > y)) &&
                (x < (poly[j].x - poly[i].x) * (y - poly[i].y) / (float)(poly[j].y - poly[i].y) + poly[i].x))
            {
                inside = !inside;
            }
        }
        return inside;
    }

   private void RecalculateBounds(int w, int h)
    {
        int minX = w, maxX = -1, minY = h, maxY = -1;
        bool any = false;

        for (int x = 0; x < w; x++)
        {
            for (int y = 0; y < h; y++)
            {
                if (mask[x, y])
                {
                    any = true;
                    if (x < minX) minX = x;
                    if (x > maxX) maxX = x;
                    if (y < minY) minY = y;
                    if (y > maxY) maxY = y;
                }
            }
        }

        HasSelection = any;
        if (any)
        {
            Bounds = new Rectangle(minX, minY, maxX - minX + 1, maxY - minY + 1);
        }
        else
        {
            Bounds = new Rectangle(0, 0, 0, 0);
        }
    }

    public void StartMove(Canvas canvas, HistoryManager history, int gx, int gy)
    {
        if (!HasSelection) return;

        if (!IsFloating)
        {
            history.SaveState(canvas);
            LiftSelection(canvas);
        }

        IsDraggingFloating = true;
        dragStartGrid = (gx, gy);
        initialOffset = Offset;
    }

    public void UpdateMove(int gx, int gy)
    {
        if (!IsDraggingFloating) return;
        int dx = gx - dragStartGrid.x;
        int dy = gy - dragStartGrid.y;
        Offset = new Vector2(initialOffset.X + dx, initialOffset.Y + dy);
    }

    public void EndMove()
    {
        IsDraggingFloating = false;
    }

    private void LiftSelection(Canvas canvas)
    {
        if (IsFloating) return;
        int w = canvas.Width;
        int h = canvas.Height;
        EnsureSize(w, h);

        for (int x = 0; x < w; x++)
        {
            for (int y = 0; y < h; y++)
            {
                if (mask[x, y])
                {
                    floatingPixels[x, y] = canvas.GetPixel(x, y);
                    canvas.SetPixel(x, y, Canvas.TransparentColor);
                }
                else
                {
                    floatingPixels[x, y] = Canvas.TransparentColor;
                }
            }
        }
        IsFloating = true;
    }

    public void CancelOrCommit(Canvas canvas)
    {
        if (!HasSelection && !IsFloating) return;

        if (IsFloating)
        {
            int w = canvas.Width;
            int h = canvas.Height;
            int ox = (int)Offset.X;
            int oy = (int)Offset.Y;

            for (int x = 0; x < w; x++)
            {
                for (int y = 0; y < h; y++)
                {
                    if (mask[x, y])
                    {
                        Color c = floatingPixels[x, y];
                        if (c.A > 0)
                        {
                            int tx = x + ox;
                            int ty = y + oy;
                            if (tx >= 0 && tx < w && ty >= 0 && ty < h)
                            {
                                canvas.SetPixel(tx, ty, c);
                            }
                        }
                    }
                }
            }
        }

        IsFloating = false;
        HasSelection = false;
        IsDraggingFloating = false;
        IsCreatingSelection = false;
        Offset = Vector2.Zero;
        Array.Clear(mask, 0, mask.Length);
    }

    public void DeleteSelection(Canvas canvas, HistoryManager history)
    {
        if (!HasSelection) return;

        if (!IsFloating)
        {
            history.SaveState(canvas);
            int w = canvas.Width;
            int h = canvas.Height;
            for (int x = 0; x < w; x++)
            {
                for (int y = 0; y < h; y++)
                {
                    if (mask[x, y])
                    {
                        canvas.SetPixel(x, y, Canvas.TransparentColor);
                    }
                }
            }
        }

        IsFloating = false;
        HasSelection = false;
        IsDraggingFloating = false;
        Offset = Vector2.Zero;
        Array.Clear(mask, 0, mask.Length);
    }

    public void Copy(Canvas canvas)
    {
        if (!HasSelection) return;

        int bx = (int)Bounds.X;
        int by = (int)Bounds.Y;
        clipWidth = (int)Bounds.Width;
        clipHeight = (int)Bounds.Height;

        clipboardPixels = new Color[clipWidth, clipHeight];
        clipboardMask = new bool[clipWidth, clipHeight];

        for (int x = 0; x < clipWidth; x++)
        {
            for (int y = 0; y < clipHeight; y++)
            {
                int sx = bx + x;
                int sy = by + y;
                if (sx >= 0 && sx < mask.GetLength(0) && sy >= 0 && sy < mask.GetLength(1) && mask[sx, sy])
                {
                    clipboardMask[x, y] = true;
                    clipboardPixels[x, y] = IsFloating ? floatingPixels[sx, sy] : canvas.GetPixel(sx, sy);
                }
            }
        }
    }

    public void Cut(Canvas canvas, HistoryManager history)
    {
        if (!HasSelection) return;
        Copy(canvas);
        DeleteSelection(canvas, history);
    }

    public void Paste(Canvas canvas, HistoryManager history)
    {
        if (clipboardPixels == null || clipboardMask == null) return;

        CancelOrCommit(canvas);
        history.SaveState(canvas);

        int w = canvas.Width;
        int h = canvas.Height;
        EnsureSize(w, h);
        Array.Clear(mask, 0, mask.Length);
        Array.Clear(floatingPixels, 0, floatingPixels.Length);

        for (int x = 0; x < clipWidth && x < w; x++)
        {
            for (int y = 0; y < clipHeight && y < h; y++)
            {
                if (clipboardMask[x, y])
                {
                    mask[x, y] = true;
                    floatingPixels[x, y] = clipboardPixels[x, y];
                }
            }
        }

        IsFloating = true;
        Offset = new Vector2(1, 1);
        RecalculateBounds(w, h);
    }

public class ToolManager
{
    public int PenSize { get; set; } = 1;
    public int EraserSize { get; set; } = 1;

    public ToolType ActiveTool { get; set; } = ToolType.Pen;
    public ShapeType ActiveShape { get; set; } = ShapeType.RectangleOutline;
    public SelectToolMode ActiveSelectionMode { get; set; } = SelectToolMode.Rectangle;

    public void ApplyTool(Canvas canvas, int startX, int startY, Color color)
    {
        switch (ActiveTool)
        {
            case ToolType.Pen:
            case ToolType.Eraser:
                int size = ActiveTool == ToolType.Pen ? PenSize : EraserSize;
                int offset = size / 2;

                for (int i = 0; i < size; i++)
                {
                    for (int j = 0; j < size; j++)
                    {
                        int px = startX - offset + i;
                        int py = startY - offset + j;
                        
                        if (ActiveTool == ToolType.Pen)
                            canvas.SetPixel(px, py, color);
                        else
                            canvas.SetPixel(px, py, Canvas.TransparentColor);
                    }
                }
                break;

            case ToolType.Bucket:
                FloodFill(canvas, startX, startY, color);
                break;

            case ToolType.SmartBucket:
                ReplaceAllColor(canvas, startX, startY, color);
                break;
        }
    }

    public void ReplaceAllColor(Canvas canvas, int startX, int startY, Color newColor)
    {
        Color targetColor = canvas.GetPixel(startX, startY);
        if (ColorsEqual(targetColor, newColor)) return;

        for (int x = 0; x < canvas.Width; x++)
        {
            for (int y = 0; y < canvas.Height; y++)
            {
                if (ColorsEqual(canvas.GetPixel(x, y), targetColor))
                {
                    canvas.SetPixel(x, y, newColor);
                }
            }
        }
    }

    public void DrawLine(Canvas canvas, int x0, int y0, int x1, int y1, Color color)
    {
        // Fareyi hÄ±zlÄ± sÃ¼rÃ¼klerken oluÅŸan Ã§izgilerde de fÄ±rÃ§a boyutunu uygular
        int size = ActiveTool == ToolType.Pen ? PenSize : EraserSize;
        int offset = size / 2;

        foreach (var (x, y) in GetLinePixels(x0, y0, x1, y1))
        {
            // Sadece tek piksel deÄŸil, seÃ§ili boyuta gÃ¶re blok halinde Ã§izer

    public static List<(int x, int y)> GetLinePixels(int x0, int y0, int x1, int y1)
    {
        List<(int x, int y)> list = new();
        int dx = Math.Abs(x1 - x0);
        int dy = Math.Abs(y1 - y0);
        int sx = x0 < x1 ? 1 : -1;
        int sy = y0 < y1 ? 1 : -1;
        int err = dx - dy;

        while (true)
        {
            list.Add((x0, y0));
            if (x0 == x1 && y0 == y1) break;

            int e2 = 2 * err;
            if (e2 > -dy)
            {
                err -= dy;
                x0 += sx;
            }
            if (e2 < dx)
            {
                err += dx;
                y0 += sy;
            }
        }
        return list;
    }

    public static List<(int x, int y)> GetShapePixels(int x0, int y0, int x1, int y1, ShapeType shape)
    {
        List<(int x, int y)> pixels = new();
        int minX = Math.Min(x0, x1);
        int maxX = Math.Max(x0, x1);
        int minY = Math.Min(y0, y1);
        int maxY = Math.Max(y0, y1);

        switch (shape)
        {
            case ShapeType.RectangleOutline:
                for (int x = minX; x <= maxX; x++) { pixels.Add((x, minY)); pixels.Add((x, maxY)); }
                for (int y = minY; y <= maxY; y++) { pixels.Add((minX, y)); pixels.Add((maxX, y)); }
                break;
            case ShapeType.RectangleFilled:
                for (int x = minX; x <= maxX; x++)
                    for (int y = minY; y <= maxY; y++)
                        pixels.Add((x, y));
                break;
            case ShapeType.CircleOutline:
            case ShapeType.CircleFilled:
                int rx = Math.Max(1, (maxX - minX) / 2);
                int ry = Math.Max(1, (maxY - minY) / 2);
                int cx = (minX + maxX) / 2;
                int cy = (minY + maxY) / 2;

                if (shape == ShapeType.CircleFilled)
                {
                    for (int y = -ry; y <= ry; y++)
                        for (int x = -rx; x <= rx; x++)
                            if ((x * x * ry * ry) + (y * y * rx * rx) <= (rx * rx * ry * ry))
                                pixels.Add((cx + x, cy + y));
                }
                else
                {
                    int x = 0, y = ry;
                    long rx2 = (long)rx * rx, ry2 = (long)ry * ry;
                    long twoRx2 = 2 * rx2, twoRy2 = 2 * ry2;
                    long px = 0, py = twoRx2 * y;

                    void AddSymmetric(int sx, int sy)
                    {
                        pixels.Add((cx + sx, cy + sy)); pixels.Add((cx - sx, cy + sy));
                        pixels.Add((cx + sx, cy - sy)); pixels.Add((cx - sx, cy - sy));
                    }

                    long p = (long)Math.Round(ry2 - (rx2 * ry) + (0.25 * rx2));
                    while (px < py)
                    {
                        AddSymmetric(x, y);
                        x++;
                        px += twoRy2;
                        if (p < 0) p += ry2 + px;
                        else { y--; py -= twoRx2; p += ry2 + px - py; }
                    }

                    p = (long)Math.Round(ry2 * (x + 0.5) * (x + 0.5) + rx2 * (y - 1) * (y - 1) - rx2 * ry2);
                    while (y >= 0)
                    {
                        AddSymmetric(x, y);
                        y--;
                        py -= twoRx2;
                        if (p > 0) p += rx2 - py;
                        else { x++; px += twoRy2; p += rx2 - py + px; }
                    }
                }
                break;
            case ShapeType.TriangleOutline:
                int midTopX = (minX + maxX) / 2;
                pixels.AddRange(GetLinePixels(midTopX, minY, minX, maxY));
                pixels.AddRange(GetLinePixels(minX, maxY, maxX, maxY));
                pixels.AddRange(GetLinePixels(maxX, maxY, midTopX, minY));
                break;
            case ShapeType.TriangleFilled:
                int topX = (minX + maxX) / 2;
                for (int y = minY; y <= maxY; y++)
                {
                    float factor = (maxY == minY) ? 1.0f : (float)(y - minY) / (maxY - minY);
                    int curMinX = (int)(topX - (topX - minX) * factor);
                    int curMaxX = (int)(topX + (maxX - topX) * factor);
                    for (int x = curMinX; x <= curMaxX; x++) pixels.Add((x, y));
                }
                break;
        }
        return pixels;
    }

    private void FloodFill(Canvas canvas, int startX, int startY, Color newColor)
    {
        Color targetColor = canvas.GetPixel(startX, startY);
        if (ColorsEqual(targetColor, newColor)) return;

        Queue<(int x, int y)> queue = new();
        queue.Enqueue((startX, startY));

        while (queue.Count > 0)
        {
            var (x, y) = queue.Dequeue();
            if (x < 0 || x >= canvas.Width || y < 0 || y >= canvas.Height) continue;
            if (!ColorsEqual(canvas.GetPixel(x, y), targetColor)) continue;

            canvas.SetPixel(x, y, newColor);
            queue.Enqueue((x + 1, y)); queue.Enqueue((x - 1, y));
            queue.Enqueue((x, y + 1)); queue.Enqueue((x, y - 1));
        }
    }

    public bool ColorsEqual(Color c1, Color c2)
    {
        return c1.R == c2.R && c1.G == c2.G && c1.B == c2.B && c1.A == c2.A;
    }
}

    public static bool SaveExport(Canvas canvas, string path, int scale, bool exportAllFrames, bool sheetVertical)
    {
        if (string.IsNullOrWhiteSpace(path)) return false;

        try
        {
            List<Frame> framesToExport = new();
            if (exportAllFrames)
            {
                foreach (var f in canvas.Frames) if (f.IsVisible) framesToExport.Add(f);
            }
            else
            {
                framesToExport.Add(canvas.Frames[canvas.ActiveFrameIndex]);
            }

            if (framesToExport.Count == 0) return false;

            int baseW = canvas.Width;
            int baseH = canvas.Height;
            int imgW = exportAllFrames ? (sheetVertical ? baseW : baseW * framesToExport.Count) : baseW;
            int imgH = exportAllFrames ? (sheetVertical ? baseH * framesToExport.Count : baseH) : baseH;

            int finalW = imgW * scale;
            int finalH = imgH * scale;

            Color bgColor = path.EndsWith(".jpg", StringComparison.OrdinalIgnoreCase) ? Color.White : Canvas.TransparentColor;
            Image img = Raylib.GenImageColor(finalW, finalH, bgColor);

            for (int i = 0; i < framesToExport.Count; i++)
            {
                int offsetX = (exportAllFrames && !sheetVertical) ? i * baseW * scale : 0;
                int offsetY = (exportAllFrames && sheetVertical) ? i * baseH * scale : 0;

                var frame = framesToExport[i];

                for (int x = 0; x < baseW; x++)
                {
                    for (int y = 0; y < baseH; y++)
                    {
                        Color pixelColor = Canvas.TransparentColor;
                        foreach (var l in frame.Layers)
                        {
                            if (l.IsVisible && l.Pixels[x, y].A > 0)
                                pixelColor = l.Pixels[x, y];
                        }

                        if (pixelColor.A > 0)
                        {
                            if (scale == 1) Raylib.ImageDrawPixel(ref img, offsetX + x, offsetY + y, pixelColor);
                            else Raylib.ImageDrawRectangle(ref img, offsetX + x * scale, offsetY + y * scale, scale, scale, pixelColor);
                        }
                    }
                }
            }

            Raylib.ExportImage(img, path);
            Raylib.UnloadImage(img);
            return true;
        }
        catch
        {
            return false;
        }
    }

    public bool Update(Canvas canvas, HistoryManager history, Vector2 mousePos, int screenWidth, int screenHeight, Action onFrameChanged)
    {
        if (IsPlaying)
        {
            playTimer += Raylib.GetFrameTime();
            if (playTimer >= 1f / Fps)
            {
                playTimer = 0f;
                int startIdx = canvas.ActiveFrameIndex;
                int nextIdx = startIdx;
                bool foundVisible = false;

                for (int i = 0; i < canvas.Frames.Count; i++)
                {
                    nextIdx = (nextIdx + 1) % canvas.Frames.Count;
                    if (canvas.Frames[nextIdx].IsVisible)
                    {
                        foundVisible = true;
                        break;
                    }
                }

                if (foundVisible)
                {
                    canvas.ActiveFrameIndex = nextIdx;
                    onFrameChanged();
                }
                else
                {
                    IsPlaying = false; 
                }
            }
        }

        Rectangle panelRect = GetPanelBounds(screenWidth, screenHeight);
        Rectangle playBtn = new Rectangle(panelRect.X + panelRect.Width - 120, panelRect.Y + 16, 50, 64);

        }

        if (Raylib.IsMouseButtonPressed(MouseButton.Left))

        float wheel = Raylib.GetMouseWheelMove();

        if (draggingFrameIndex != -1 && Raylib.IsMouseButtonReleased(MouseButton.Left))
        {

            dropIndex = Math.Clamp(dropIndex, 0, canvas.Frames.Count - 1);

            if (dropIndex != draggingFrameIndex)
            {
                history.SaveState(canvas);
                var frame = canvas.Frames[draggingFrameIndex];
                canvas.Frames.RemoveAt(draggingFrameIndex);
                canvas.Frames.Insert(dropIndex, frame);
                canvas.ActiveFrameIndex = dropIndex;
                onFrameChanged();
            }
            draggingFrameIndex = -1;
            return true;
        }

        if (Raylib.IsMouseButtonPressed(MouseButton.Left))
        {
            for (int i = 0; i <= canvas.Frames.Count; i++)
            {
                Rectangle boxRect = new Rectangle(startX + i * (boxSize + spacing), startY, boxSize, boxSize);
                if (boxRect.X + boxSize > panelRect.X && boxRect.X < panelRect.X + listAvailableWidth && Raylib.CheckCollisionPointRec(mousePos, boxRect))
                {
                    if (i == canvas.Frames.Count)
                    {
                        history.SaveState(canvas);
                        canvas.AddFrame();
                        scrollX = Math.Max(0, (canvas.Frames.Count + 1) * (boxSize + spacing) + 24 - listAvailableWidth);
                        onFrameChanged();
                    }
                    else
                    {
                        Rectangle eyeRect = new Rectangle(boxRect.X + boxSize - 22, boxRect.Y + boxSize - 18, 20, 16);
                        Rectangle delBtn = new Rectangle(boxRect.X + boxSize - 18, boxRect.Y + 2, 16, 16);

                        if (Raylib.CheckCollisionPointRec(mousePos, eyeRect))
                        {
                            canvas.Frames[i].IsVisible = !canvas.Frames[i].IsVisible;
                            onFrameChanged();
                        }
                        else if (Raylib.CheckCollisionPointRec(mousePos, delBtn) && canvas.Frames.Count > 1)
                        {
                            history.SaveState(canvas);
                            canvas.Frames.RemoveAt(i);
                            if (canvas.ActiveFrameIndex >= canvas.Frames.Count) canvas.ActiveFrameIndex = canvas.Frames.Count - 1;
                            onFrameChanged();
                            return true;
                        }
                        else
                        {
                            draggingFrameIndex = i;
                            dragOffsetX = mousePos.X - boxRect.X;
                            canvas.ActiveFrameIndex = i;
                            onFrameChanged();
                        }
                    }
                    return true;
                }
            }
        }
        return Raylib.IsMouseButtonPressed(MouseButton.Left) || Raylib.IsMouseButtonDown(MouseButton.Left);
    }

    private void DrawFrameBox(Canvas canvas, int index, Rectangle boxRect, Vector2 mousePos, bool isDragging = false)
    {
        int boxSize = (int)boxRect.Width;
        bool isActive = (index == canvas.ActiveFrameIndex);
        bool vis = canvas.Frames[index].IsVisible;

        for (int cX = 0; cX < boxSize; cX += 4)
            for (int cY = 0; cY < boxSize; cY += 4)
                Raylib.DrawRectangle((int)boxRect.X + cX, (int)boxRect.Y + cY, Math.Min(4, boxSize - cX), Math.Min(4, boxSize - cY), 
                    ((cX / 4 + cY / 4) % 2 == 0) ? new Color(170, 170, 170, 255) : new Color(130, 130, 130, 255));

        float scale = (float)boxSize / Math.Max(canvas.Width, canvas.Height);
        int thumbX = (int)boxRect.X + (boxSize - (int)(canvas.Width * scale)) / 2;
        int thumbY = (int)boxRect.Y + (boxSize - (int)(canvas.Height * scale)) / 2;

        var frame = canvas.Frames[index];
        for (int l = 0; l < frame.Layers.Count; l++)
        {
            if (!frame.Layers[l].IsVisible) continue;
            var pixels = frame.Layers[l].Pixels;
            for (int x = 0; x < canvas.Width; x++)
                for (int y = 0; y < canvas.Height; y++)
                    if (pixels[x, y].A > 0)
                        Raylib.DrawRectangle((int)(thumbX + x * scale), (int)(thumbY + y * scale), Math.Max(1, (int)Math.Ceiling(scale)), Math.Max(1, (int)Math.Ceiling(scale)), pixels[x, y]);
        }

        if (!vis) Raylib.DrawRectangleRec(boxRect, new Color(0, 0, 0, 160));

        Raylib.DrawRectangle((int)boxRect.X, (int)boxRect.Y, boxSize, 18, new Color(0, 0, 0, 160));
        Raylib.DrawText($"Kare {index + 1}", (int)boxRect.X + 6, (int)boxRect.Y + 4, 10, isActive ? Color.Yellow : Color.RayWhite);

        int eyeCx = (int)boxRect.X + boxSize - 12;
        int eyeCy = (int)boxRect.Y + boxSize - 10;
        if (vis) { Raylib.DrawEllipse(eyeCx, eyeCy, 6, 4, Color.White); Raylib.DrawCircle(eyeCx, eyeCy, 2, Color.Black); }
        else { Raylib.DrawLine(eyeCx - 5, eyeCy - 4, eyeCx + 5, eyeCy + 4, Color.Red); Raylib.DrawLine(eyeCx - 5, eyeCy + 4, eyeCx + 5, eyeCy - 4, Color.Red); }
        
        Color borderColor = isDragging ? Color.Yellow : (isActive ? Color.Yellow : new Color(30, 30, 30, 255));
        Raylib.DrawRectangleLinesEx(boxRect, isActive || isDragging ? 2 : 1, borderColor);

        bool isHover = Raylib.CheckCollisionPointRec(mousePos, boxRect);
        if (isHover && !isDragging && canvas.Frames.Count > 1)
        {
            Rectangle delBtn = new Rectangle(boxRect.X + boxSize - 18, boxRect.Y + 2, 16, 16);
            bool delHover = Raylib.CheckCollisionPointRec(mousePos, delBtn);
            Raylib.DrawRectangleRec(delBtn, delHover ? new Color(220, 50, 50, 255) : new Color(150, 30, 30, 200));
            Raylib.DrawRectangleLinesEx(delBtn, 1, Color.White);
            Raylib.DrawText("x", (int)delBtn.X + 5, (int)delBtn.Y + 2, 12, Color.White);
        }

        if (isDragging) Raylib.DrawRectangleRec(boxRect, new Color(255, 255, 255, 50));
    }

using System;
using System.Globalization;
using System.Numerics;
using System.Threading;
using System.Windows.Forms;
using Raylib_cs;

namespace PixelArtApp.UI;

public class ColorPickerDialog
{
    public bool IsOpen { get; private set; } = false;

    private Color initialColor;
    private Color currentColor;

    private float hue = 0f;
    private float sat = 1f;
    private float val = 1f;
    private int alpha = 255;

    private Vector2 svHandlePos;

    // HEX Metin Kutusu
    private string hexText = "";
    private bool hexInputActive = false;
    private bool isTextSelected = false;
    private float backspaceHoldTimer = 0f;

    // Pencere ve SÃ¼rÃ¼kleme DurumlarÄ±
    private Rectangle windowRect = new Rectangle(200, 150, 465, 350);
    private bool isDraggingWindow = false;

    private void InitTexture()

    public void Open(Color startingColor)
    {
        InitTexture();

        initialColor = startingColor;
        currentColor = startingColor;
        alpha = startingColor.A;

        RgbToHsv(startingColor, out hue, out sat, out val);
        svHandlePos = new Vector2(sat * 180f, (1f - val) * 180f);

        SyncHex();
        UpdateSvTexture();

        int sw = Raylib.GetScreenWidth();
        int sh = Raylib.GetScreenHeight();
        windowRect.X = (sw - windowRect.Width) / 2f;
        windowRect.Y = (sh - windowRect.Height) / 2f;

        ResetDragStates();

        ResetDragStates();

    private void ResetDragStates()
    {
        isDraggingWindow = false;
        isDraggingSV = false;
        isDraggingHue = false;
        isDraggingVal = false;
        isDraggingAlpha = false;
        hexInputActive = false;
        isTextSelected = false;
    }

    public bool Update(Vector2 mousePos, Action<Color, bool> onConfirm)
    {
        if (!IsOpen) return false;

        Rectangle titleBar = new Rectangle(windowRect.X, windowRect.Y, windowRect.Width, 32);
        Rectangle svBox = new Rectangle(windowRect.X + 20, windowRect.Y + 45, 180, 180);
        Rectangle hueBar = new Rectangle(windowRect.X + 210, windowRect.Y + 45, 20, 180);
        Rectangle valBar = new Rectangle(windowRect.X + 238, windowRect.Y + 45, 20, 180);
        Rectangle alphaBar = new Rectangle(windowRect.X + 20, windowRect.Y + 235, 180, 16);
        Rectangle alphaHitBox = new Rectangle(alphaBar.X, alphaBar.Y - 4, alphaBar.Width, 24);
        Rectangle hexBox = new Rectangle(windowRect.X + 275, windowRect.Y + 143, 160, 30);

        // 1. TIKLAMA BAÅLANGICI
        if (Raylib.IsMouseButtonPressed(MouseButton.Left))
        {
            if (Raylib.CheckCollisionPointRec(mousePos, titleBar))
            {
                isDraggingWindow = true;
                dragWindowOffset = new Vector2(mousePos.X - windowRect.X, mousePos.Y - windowRect.Y);
            }
            else if (Raylib.CheckCollisionPointRec(mousePos, svBox))
            {
                isDraggingSV = true;
                hexInputActive = false;
                isTextSelected = false;
            }
            else if (Raylib.CheckCollisionPointRec(mousePos, hueBar))
            {
                isDraggingHue = true;
                hexInputActive = false;
                isTextSelected = false;
            }
            else if (Raylib.CheckCollisionPointRec(mousePos, valBar))
            {
                isDraggingVal = true;
                hexInputActive = false;
                isTextSelected = false;
            }
            else if (Raylib.CheckCollisionPointRec(mousePos, alphaHitBox))
            {
                isDraggingAlpha = true;
                hexInputActive = false;
                isTextSelected = false;
            }
            else
            {
                bool clickedHex = Raylib.CheckCollisionPointRec(mousePos, hexBox);

                if (hexInputActive && !clickedHex)
                {
                    if (hexText.Length > 0 && hexText.Length != 3 && hexText.Length < 6)
                    {
                        hexText = hexText.PadRight(6, '0');
                    }
                    ApplyHex(true);
                    SyncHex();
                }

                if (clickedHex && !hexInputActive) isTextSelected = true;
                else if (!clickedHex) isTextSelected = false;
                hexInputActive = clickedHex;
            }
        }

        // 2. SÃœRÃœKLEME
        if (Raylib.IsMouseButtonDown(MouseButton.Left))
        {
            if (isDraggingWindow)
            {
                windowRect.X = Math.Clamp(mousePos.X - dragWindowOffset.X, 0, Raylib.GetScreenWidth() - windowRect.Width);
                windowRect.Y = Math.Clamp(mousePos.Y - dragWindowOffset.Y, 0, Raylib.GetScreenHeight() - windowRect.Height);
            }
            else if (isDraggingSV)
            {
                svHandlePos.X = Math.Clamp(mousePos.X - svBox.X, 0, 180);
                svHandlePos.Y = Math.Clamp(mousePos.Y - svBox.Y, 0, 180);
                sat = svHandlePos.X / 180f;
                val = 1f - (svHandlePos.Y / 180f);
                currentColor = HsvToRgb(hue, sat, val, alpha);
                SyncHex();
            }
            else if (isDraggingHue)
            {
                float relativeY = Math.Clamp(mousePos.Y - hueBar.Y, 0, 180);
                float newHue = (relativeY / 180f) * 360f;
                if (newHue >= 360f) newHue = 359.9f;

                if (Math.Abs(newHue - hue) > 0.5f)
                {
                    hue = newHue;
                    UpdateSvTexture();
                    currentColor = HsvToRgb(hue, sat, val, alpha);
                    SyncHex();
                }
            }
            else if (isDraggingVal)
            {
                // YukarÄ±dan aÅŸaÄŸÄ± Ã§ekildikÃ§e parlaklÄ±ÄŸÄ± (val) 1.0 -> 0.0 arasÄ± deÄŸiÅŸtirir
                float relativeY = Math.Clamp(mousePos.Y - valBar.Y, 0, 180);
                val = 1f - (relativeY / 180f);
                svHandlePos.Y = relativeY;
                currentColor = HsvToRgb(hue, sat, val, alpha);
                SyncHex();
            }
            else if (isDraggingAlpha)
            {
                float alphaRatio = Math.Clamp((mousePos.X - alphaBar.X) / alphaBar.Width, 0f, 1f);
                alpha = (int)Math.Round(alphaRatio * 255f);
                currentColor.A = (byte)alpha;
            }
        }
        else
        {
            isDraggingWindow = false;
            isDraggingSV = false;
            isDraggingHue = false;
            isDraggingVal = false;
            isDraggingAlpha = false;
        }

        // 3. HEX Klavye Girdisi
        if (hexInputActive)
        {
            bool isCtrl = Raylib.IsKeyDown(KeyboardKey.LeftControl) || Raylib.IsKeyDown(KeyboardKey.RightControl);

            if (isCtrl && Raylib.IsKeyPressed(KeyboardKey.A)) isTextSelected = true;
            if (isCtrl && Raylib.IsKeyPressed(KeyboardKey.C)) SetClipboardSafe($"#{hexText}");
            if (isCtrl && Raylib.IsKeyPressed(KeyboardKey.V))
            {
                string clipboard = GetClipboardSafe();
                if (!string.IsNullOrEmpty(clipboard))
                {
                    clipboard = clipboard.Trim().TrimStart('#');
                    string filtered = "";
                    foreach (char c in clipboard)
                    {
                        if (char.IsAsciiHexDigit(c) && filtered.Length < 6) filtered += char.ToUpper(c);
                    }
                    if (filtered.Length > 0)
                    {
                        if (filtered.Length != 3 && filtered.Length < 6)
                        {
                            filtered = filtered.PadRight(6, filtered[^1]);
                        }

                        hexText = filtered;
                        isTextSelected = false;
                        ApplyHex(true);
                    }
                }
            }

            int key = Raylib.GetCharPressed();
            while (key > 0)
            {
                char c = (char)key;
                if (char.IsAsciiHexDigit(c))
                {
                    if (isTextSelected)
                    {
                        hexText = "";
                        isTextSelected = false;
                    }
                    if (hexText.Length < 6)
                    {
                        hexText += char.ToUpper(c);
                        ApplyHex();
                    }
                }
                key = Raylib.GetCharPressed();
            }

            if (Raylib.IsKeyPressed(KeyboardKey.Backspace))
            {
                if (isTextSelected)
                {
                    hexText = "";
                    isTextSelected = false;
                    ApplyHex();
                }
                else if (hexText.Length > 0)
                {
                    hexText = hexText[..^1];
                    backspaceHoldTimer = 0.35f;
                    ApplyHex();
                }
            }
            else if (Raylib.IsKeyDown(KeyboardKey.Backspace) && !isTextSelected && hexText.Length > 0)
            {
                backspaceHoldTimer -= Raylib.GetFrameTime();
                if (backspaceHoldTimer <= 0)
                {
                    hexText = hexText[..^1];
                    backspaceHoldTimer = 0.05f;
                    ApplyHex();
                }
            }

            if (Raylib.IsKeyPressed(KeyboardKey.Enter) || Raylib.IsKeyPressed(KeyboardKey.Escape))
            {
                hexInputActive = false;
                isTextSelected = false;

                if (hexText.Length > 0 && hexText.Length != 3 && hexText.Length < 6)
                {
                    hexText = hexText.PadRight(6, '0');
                }
                ApplyHex(true);
                SyncHex();
            }
        }

        // 4. BUTONLAR
        Rectangle btnApply = new Rectangle(windowRect.X + 20, windowRect.Y + 295, 115, 34);
        Rectangle btnAddPalette = new Rectangle(windowRect.X + 145, windowRect.Y + 295, 160, 34);
        Rectangle btnCancel = new Rectangle(windowRect.X + 315, windowRect.Y + 295, 125, 34);
        Rectangle btnClose = new Rectangle(windowRect.X + windowRect.Width - 28, windowRect.Y + 4, 24, 24);

        if (Raylib.IsMouseButtonPressed(MouseButton.Left))
        {
            if (Raylib.CheckCollisionPointRec(mousePos, btnApply))
            {
                onConfirm(currentColor, false);
                Close();
            }
            else if (Raylib.CheckCollisionPointRec(mousePos, btnAddPalette))
            {
                onConfirm(currentColor, true);
                Close();
            }
            else if (Raylib.CheckCollisionPointRec(mousePos, btnCancel) || Raylib.CheckCollisionPointRec(mousePos, btnClose))
            {
                Close();
            }
        }

        return true;
    }

    private string GetClipboardSafe()
    {
        string clipboardText = "";
        try
        {
            Thread staThread = new Thread(() =>
            {
                try
                {
                    if (System.Windows.Forms.Clipboard.ContainsText())
                    {
                        clipboardText = System.Windows.Forms.Clipboard.GetText();
                    }
                }
                catch { }
            });
            staThread.SetApartmentState(ApartmentState.STA);
            staThread.Start();
            staThread.Join(200);
        }
        catch { }
        return clipboardText;
    }

    private void SetClipboardSafe(string text)
    {
        try
        {
            Thread staThread = new Thread(() =>
            {
                try { System.Windows.Forms.Clipboard.SetText(text); } catch { }
            });
            staThread.SetApartmentState(ApartmentState.STA);
            staThread.Start();
            staThread.Join(200);
        }
        catch { }
    }

    private void UpdateSvTexture()
    {
        Raylib.BeginTextureMode(svTexture);
        Color pureHue = HsvToRgb(hue, 1f, 1f, 255);
        Raylib.DrawRectangleGradientH(0, 0, 180, 180, Color.White, pureHue);
        Raylib.DrawRectangleGradientV(0, 0, 180, 180, Color.Blank, Color.Black);
        Raylib.EndTextureMode();
    }

    private void SyncHex()
    {
        hexText = $"{currentColor.R:X2}{currentColor.G:X2}{currentColor.B:X2}";
    }

    private void ApplyHex(bool forceSpectrumUpdate = false)
    {
        string parseStr = hexText;

        if (parseStr.Length == 0)
        {
            parseStr = "000000";
        }
        else if (parseStr.Length == 3)
        {
            parseStr = $"{parseStr[0]}{parseStr[0]}{parseStr[1]}{parseStr[1]}{parseStr[2]}{parseStr[2]}";
        }
        else if (parseStr.Length < 6)
        {
            parseStr = parseStr.PadRight(6, '0');
        }

        if (uint.TryParse(parseStr, NumberStyles.HexNumber, null, out uint valHex))
        {
            int r = (int)((valHex >> 16) & 0xFF);
            int g = (int)((valHex >> 8) & 0xFF);
            int b = (int)(valHex & 0xFF);

            currentColor = new Color((byte)r, (byte)g, (byte)b, (byte)alpha);

            if (forceSpectrumUpdate || hexText.Length == 3 || hexText.Length == 6)
            {
                RgbToHsv(currentColor, out float newHue, out sat, out this.val);

                if (sat > 0.001f)
                {
                    hue = newHue;
                }

                svHandlePos = new Vector2(sat * 180f, (1f - this.val) * 180f);
                UpdateSvTexture();
            }
        }
    }

    private static void RgbToHsv(Color color, out float h, out float s, out float v)
    {
        float r = color.R / 255f;
        float g = color.G / 255f;
        float b = color.B / 255f;

        float max = Math.Max(r, Math.Max(g, b));
        float min = Math.Min(r, Math.Min(g, b));
        float delta = max - min;

        v = max;
        s = max == 0 ? 0 : delta / max;

        if (delta == 0) h = 0;
        else if (max == r) h = 60 * (((g - b) / delta) % 6);
        else if (max == g) h = 60 * (((b - r) / delta) + 2);
        else h = 60 * (((r - g) / delta) + 4);

        if (h < 0) h += 360f;
    }

    private static Color HsvToRgb(float h, float s, float v, int a)
    {
        float c = v * s;
        float x = c * (1 - Math.Abs((h / 60f % 2) - 1));
        float m = v - c;

        float r = 0, g = 0, b = 0;
        if (h < 60) { r = c; g = x; b = 0; }
        else if (h < 120) { r = x; g = c; b = 0; }
        else if (h < 180) { r = 0; g = c; b = x; }
        else if (h < 240) { r = 0; g = x; b = c; }
        else if (h < 300) { r = x; g = 0; b = c; }
        else { r = c; g = 0; b = x; }

        return new Color((byte)((r + m) * 255), (byte)((g + m) * 255), (byte)((b + m) * 255), (byte)a);
    }

    private float scrollDragStartScrollX = 0f;

    private int draggingLayerIndex = -1;
    private float dragOffsetX = 0f;

    private Rectangle GetPanelBounds(int screenWidth, int screenHeight)
    {
        float leftMargin = 155f;  
        float rightMargin = 190f; 
        float panelWidth = screenWidth - leftMargin - rightMargin;
        float panelY = IsHidden ? screenHeight : screenHeight - Height - 12f; 
        return new Rectangle(leftMargin, panelY, panelWidth, Height);
    }

    public bool Update(Canvas canvas, HistoryManager history, Vector2 mousePos, int screenWidth, int screenHeight, Action onLayerChanged)
    {
        Rectangle panelRect = GetPanelBounds(screenWidth, screenHeight);
        
        Rectangle toggleBtn = new Rectangle(panelRect.X + 10, panelRect.Y - 16, 40, 16);
        if (Raylib.CheckCollisionPointRec(mousePos, toggleBtn) && Raylib.IsMouseButtonPressed(MouseButton.Left))
        {
            IsHidden = !IsHidden;
            return true; 
        }

        if (IsHidden) return false;

        if (!Raylib.CheckCollisionPointRec(mousePos, panelRect)) 
        {
            if (draggingLayerIndex != -1 && Raylib.IsMouseButtonReleased(MouseButton.Left)) draggingLayerIndex = -1;
            return false;
        }

        int boxSize = 58; 
        int spacing = 10;
        float startX = panelRect.X + 12 - scrollX;
        float startY = panelRect.Y + 10;

        int itemCount = canvas.Layers.Count + 1; 
        float totalWidth = itemCount * boxSize + (itemCount - 1) * spacing + 24;
        float maxScroll = Math.Max(0, totalWidth - panelRect.Width);

        float wheel = Raylib.GetMouseWheelMove();
        if (wheel != 0) { scrollX = Math.Clamp(scrollX - wheel * 40f, 0, maxScroll); }

        if (maxScroll > 0)
        {
            float trackX = panelRect.X + 12;
            float trackY = panelRect.Y + Height - 12;
            float trackWidth = panelRect.Width - 24;
            float thumbWidth = Math.Max(40f, (panelRect.Width / totalWidth) * trackWidth);
            float thumbX = trackX + (scrollX / maxScroll) * (trackWidth - thumbWidth);
            Rectangle thumbRect = new Rectangle(thumbX, trackY, thumbWidth, 6);

            if (Raylib.IsMouseButtonPressed(MouseButton.Left) && Raylib.CheckCollisionPointRec(mousePos, thumbRect))
            {
                isDraggingScrollbar = true; scrollDragStartMouseX = mousePos.X; scrollDragStartScrollX = scrollX;
            }

            if (isDraggingScrollbar)
            {
                if (Raylib.IsMouseButtonDown(MouseButton.Left))
                {
                    float deltaX = mousePos.X - scrollDragStartMouseX;
                    scrollX = Math.Clamp(scrollDragStartScrollX + (deltaX / (trackWidth - thumbWidth)) * maxScroll, 0, maxScroll);
                }
                else isDraggingScrollbar = false;
                return true;
            }
        }

        if (draggingLayerIndex != -1 && Raylib.IsMouseButtonReleased(MouseButton.Left))
        {
            int dropIndex = (int)Math.Round((mousePos.X - startX) / (boxSize + spacing));
            dropIndex = Math.Clamp(dropIndex, 0, canvas.Layers.Count - 1);

            if (dropIndex != draggingLayerIndex)
            {
                history.SaveState(canvas);
                var layer = canvas.Layers[draggingLayerIndex];
                canvas.Layers.RemoveAt(draggingLayerIndex);
                canvas.Layers.Insert(dropIndex, layer);
                canvas.ActiveLayerIndex = dropIndex;
                onLayerChanged();
            }
            draggingLayerIndex = -1;
            return true;
        }

        if (Raylib.IsMouseButtonPressed(MouseButton.Left) && !isDraggingScrollbar)
        {
            for (int i = 0; i <= canvas.Layers.Count; i++)
            {
                Rectangle boxRect = new Rectangle(startX + i * (boxSize + spacing), startY, boxSize, boxSize);
                
                if (boxRect.X + boxSize > panelRect.X && boxRect.X < panelRect.X + panelRect.Width && Raylib.CheckCollisionPointRec(mousePos, boxRect))
                {
                    if (i == canvas.Layers.Count)
                    {
                        history.SaveState(canvas);
                        canvas.AddLayer();
                        scrollX = Math.Max(0, (canvas.Layers.Count + 1) * (boxSize + spacing) + 24 - panelRect.Width);
                        onLayerChanged();
                    }
                    else
                    {
                        // DÃœZELTÄ°LMÄ°Å GÃ–Z HÄ°TBOX'I (SaÄŸ Alt KÃ¶ÅŸe)
                        Rectangle eyeRect = new Rectangle(boxRect.X + boxSize - 22, boxRect.Y + boxSize - 18, 20, 16);
                        Rectangle delBtn = new Rectangle(boxRect.X + boxSize - 18, boxRect.Y + 2, 16, 16);

                        if (Raylib.CheckCollisionPointRec(mousePos, eyeRect))
                        {
                            canvas.Layers[i].IsVisible = !canvas.Layers[i].IsVisible; 
                        }
                        else if (Raylib.CheckCollisionPointRec(mousePos, delBtn) && canvas.Layers.Count > 1)
                        {
                            history.SaveState(canvas);
                            canvas.Layers.RemoveAt(i);
                            if (canvas.ActiveLayerIndex >= canvas.Layers.Count) canvas.ActiveLayerIndex = canvas.Layers.Count - 1;
                            onLayerChanged();
                            return true;
                        }
                        else
                        {
                            draggingLayerIndex = i;
                            dragOffsetX = mousePos.X - boxRect.X;
                            canvas.ActiveLayerIndex = i; 
                            onLayerChanged();
                        }
                    }
                    return true;
                }
            }
        }

        return Raylib.IsMouseButtonPressed(MouseButton.Left) || Raylib.IsMouseButtonDown(MouseButton.Left);
    }

    public void Draw(Canvas canvas, int screenWidth, int screenHeight)
    {
        Rectangle panelRect = GetPanelBounds(screenWidth, screenHeight);

        Rectangle toggleBtn = new Rectangle(panelRect.X + 10, panelRect.Y - 16, 40, 16);
        Raylib.DrawRectangleRec(toggleBtn, new Color(30, 30, 35, 255));
        Raylib.DrawRectangleLinesEx(toggleBtn, 1, new Color(70, 70, 80, 255));

        int cx = (int)toggleBtn.X + 20;
        int cy = (int)toggleBtn.Y + 8;

        if (IsHidden)
        {
            for (int j = 0; j < 2; j++) { Raylib.DrawLine(cx - 4, cy + 1 + j, cx, cy - 3 + j, Color.LightGray); Raylib.DrawLine(cx, cy - 3 + j, cx + 4, cy + 1 + j, Color.LightGray); }
            return;
        }
        else
        {
            for (int j = 0; j < 2; j++) { Raylib.DrawLine(cx - 4, cy - 2 + j, cx, cy + 2 + j, Color.LightGray); Raylib.DrawLine(cx, cy + 2 + j, cx + 4, cy - 2 + j, Color.LightGray); }
        }

        int boxSize = 58;
        int spacing = 10;
        float startX = panelRect.X + 12 - scrollX;
        float startY = panelRect.Y + 10;
        Vector2 mousePos = Raylib.GetMousePosition();

        Raylib.BeginScissorMode((int)panelRect.X, (int)panelRect.Y, (int)panelRect.Width, (int)Height);

        for (int i = 0; i <= canvas.Layers.Count; i++)
        {
            if (i == draggingLayerIndex) continue;
            Rectangle boxRect = new Rectangle(startX + i * (boxSize + spacing), startY, boxSize, boxSize);

            if (i == canvas.Layers.Count)
            {
                Raylib.DrawRectangleRec(boxRect, new Color(40, 40, 45, 255));
                Raylib.DrawRectangleLinesEx(boxRect, 1, Color.Gray);
                int fontSize = 32;
                int plusWidth = Raylib.MeasureText("+", fontSize);
                Raylib.DrawText("+", (int)boxRect.X + (boxSize - plusWidth) / 2, (int)boxRect.Y + (boxSize - fontSize) / 2, fontSize, Color.DarkGray);
            }
            else DrawLayerBox(canvas, i, boxRect, mousePos);
        }

        if (draggingLayerIndex != -1 && draggingLayerIndex < canvas.Layers.Count)
        {
            Rectangle dragRect = new Rectangle(mousePos.X - dragOffsetX, startY, boxSize, boxSize);
            int dropIndex = Math.Clamp((int)Math.Round((mousePos.X - startX) / (boxSize + spacing)), 0, canvas.Layers.Count - 1);
            Raylib.DrawRectangle((int)(startX + dropIndex * (boxSize + spacing) - (spacing / 2f)) - 2, (int)startY, 4, boxSize, Color.Yellow);
            DrawLayerBox(canvas, draggingLayerIndex, dragRect, mousePos, isDragging: true);
        }
        
        Raylib.EndScissorMode();

        float maxScroll = Math.Max(0, (canvas.Layers.Count + 1) * boxSize + (canvas.Layers.Count) * spacing + 24 - panelRect.Width);
        if (maxScroll > 0)
        {
            float trackX = panelRect.X + 12;
            float trackY = panelRect.Y + Height - 12;
            float trackWidth = panelRect.Width - 24;
            float thumbWidth = Math.Max(40f, (panelRect.Width / maxScroll) * trackWidth);
            Raylib.DrawRectangle((int)trackX, (int)trackY, (int)trackWidth, 5, new Color(15, 15, 15, 150));
            Raylib.DrawRectangle((int)(trackX + (scrollX / maxScroll) * (trackWidth - thumbWidth)), (int)trackY, (int)thumbWidth, 5, isDraggingScrollbar ? Color.SkyBlue : Color.Gray);
        }
    }

    private void DrawLayerBox(Canvas canvas, int index, Rectangle boxRect, Vector2 mousePos, bool isDragging = false)
    {
        int boxSize = (int)boxRect.Width;
        bool isActive = (index == canvas.ActiveLayerIndex);
        bool vis = canvas.Layers[index].IsVisible;

        for (int cX = 0; cX < boxSize; cX += 4)
            for (int cY = 0; cY < boxSize; cY += 4)
                Raylib.DrawRectangle((int)boxRect.X + cX, (int)boxRect.Y + cY, Math.Min(4, boxSize - cX), Math.Min(4, boxSize - cY), 
                    ((cX / 4 + cY / 4) % 2 == 0) ? new Color(170, 170, 170, 255) : new Color(130, 130, 130, 255));

        float scale = (float)boxSize / Math.Max(canvas.Width, canvas.Height);
        int thumbX = (int)boxRect.X + (boxSize - (int)(canvas.Width * scale)) / 2;
        int thumbY = (int)boxRect.Y + (boxSize - (int)(canvas.Height * scale)) / 2;

        var pixels = canvas.Layers[index].Pixels;
        for (int x = 0; x < canvas.Width; x++)
            for (int y = 0; y < canvas.Height; y++)
                if (pixels[x, y].A > 0)
                    Raylib.DrawRectangle((int)(thumbX + x * scale), (int)(thumbY + y * scale), Math.Max(1, (int)Math.Ceiling(scale)), Math.Max(1, (int)Math.Ceiling(scale)), pixels[x, y]);

        // Gizli ise grileÅŸtir
        if (!vis) Raylib.DrawRectangleRec(boxRect, new Color(0, 0, 0, 150));

        Raylib.DrawRectangle((int)boxRect.X, (int)boxRect.Y, boxSize, 18, new Color(0, 0, 0, 160));
        Raylib.DrawText((index + 1).ToString(), (int)boxRect.X + 6, (int)boxRect.Y + 4, 11, isActive ? Color.Yellow : Color.RayWhite);

        // DÃœZELTÄ°LMÄ°Å GÃ–Z Ã‡Ä°ZÄ°MÄ° (SaÄŸ Alt KÃ¶ÅŸe)
        int eyeCx = (int)boxRect.X + boxSize - 12;
        int eyeCy = (int)boxRect.Y + boxSize - 10;
        if (vis)
        {
            Raylib.DrawEllipse(eyeCx, eyeCy, 6, 4, Color.White);
            Raylib.DrawCircle(eyeCx, eyeCy, 2, Color.Black);
        }
        else
        {
            Raylib.DrawLine(eyeCx - 5, eyeCy - 4, eyeCx + 5, eyeCy + 4, Color.Red);
            Raylib.DrawLine(eyeCx - 5, eyeCy + 4, eyeCx + 5, eyeCy - 4, Color.Red);
        }

        Color borderColor = isDragging ? Color.Yellow : (isActive ? Color.SkyBlue : new Color(30, 30, 30, 255));
        Raylib.DrawRectangleLinesEx(boxRect, isActive || isDragging ? 2 : 1, borderColor);

        bool isHover = Raylib.CheckCollisionPointRec(mousePos, boxRect);
        if (isHover && !isDragging && canvas.Layers.Count > 1)
        {
            Rectangle delBtn = new Rectangle(boxRect.X + boxSize - 18, boxRect.Y + 2, 16, 16);
            bool delHover = Raylib.CheckCollisionPointRec(mousePos, delBtn);
            Raylib.DrawRectangleRec(delBtn, delHover ? new Color(220, 50, 50, 255) : new Color(150, 30, 30, 200));
            Raylib.DrawRectangleLinesEx(delBtn, 1, Color.White);
            Raylib.DrawText("x", (int)delBtn.X + 5, (int)delBtn.Y + 2, 12, Color.White);
        }

        if (isDragging) Raylib.DrawRectangleRec(boxRect, new Color(255, 255, 255, 50));
    }
}

    public bool Update(Vector2 mousePos, Action<int, int, Color> onConfirm)

        if (Raylib.IsMouseButtonPressed(MouseButton.Left))
        {
            if (Raylib.CheckCollisionPointRec(mousePos, wBox)) activeInput = 1;
            else if (Raylib.CheckCollisionPointRec(mousePos, hBox)) activeInput = 2;
            else activeInput = 0;

            // HazÄ±r Boyut ÅablonlarÄ±

                Rectangle pRect = new Rectangle(windowRect.X + 215 + (i % 2) * 62, windowRect.Y + 65 + (i / 2) * 34, 56, 28);
                if (Raylib.CheckCollisionPointRec(mousePos, pRect))

                    widthText = presets[i].ToString();
                    heightText = presets[i].ToString();
                }
            }

            // Background SeÃ§enekleri
            for (int i = 0; i < 3; i++)
            {
                Rectangle bgRect = new Rectangle(windowRect.X + 25 + i * 105, windowRect.Y + 175, 95, 58);
                if (Raylib.CheckCollisionPointRec(mousePos, bgRect))

            if (Raylib.CheckCollisionPointRec(mousePos, okBtn))

                ConfirmCreation(onConfirm);
                return true;
            }
            else if (Raylib.CheckCollisionPointRec(mousePos, cancelBtn))
            {
                Close();
                return true;
            }
        }

        // Klavyeden Rakam GiriÅŸi
        if (activeInput != 0)
        {
            int key = Raylib.GetCharPressed();
            while (key > 0)
            {
                if (char.IsDigit((char)key))
                {
                    ref string target = ref (activeInput == 1 ? ref widthText : ref heightText);
                    if (target.Length < 3) target += (char)key;
                }
                key = Raylib.GetCharPressed();
            }

            if (Raylib.IsKeyPressed(KeyboardKey.Backspace))
            {
                ref string target = ref (activeInput == 1 ? ref widthText : ref heightText);
                if (target.Length > 0) target = target[..^1];
            }

            if (Raylib.IsKeyPressed(KeyboardKey.Tab))
            {
                activeInput = activeInput == 1 ? 2 : 1;
            }
        }

        if (Raylib.IsKeyPressed(KeyboardKey.Enter))
        {
            ConfirmCreation(onConfirm);
        }
        else if (Raylib.IsKeyPressed(KeyboardKey.Escape))
        {
            Close();
        }

        return true;
    }

        {
            BgOption.White => Color.White,
            BgOption.Black => Color.Black,
            _ => Canvas.TransparentColor
        };

        onConfirm(w, h, bg);
        Close();
    }

using System;

public class Palette
{
    public Color CurrentColor { get; set; } = Color.Black;
    public Vector2 Offset { get; set; }

    private readonly List<Color> colors = new();
    private readonly int boxSize = 22;
    private readonly int spacing = 5;
    private readonly int columns = 5;

    // KaydÄ±rma & Ã‡erÃ§eve Parametreleri
    private float scrollY = 0f;
    private bool isDraggingScrollbar = false;
    private float scrollDragStartMouseY = 0f;
    private float scrollDragStartScrollY = 0f;

    private readonly float viewportWidth = 144f;
    private readonly float viewportHeight = 250f;

    public Rectangle ActiveColorBoxBounds => new Rectangle(Offset.X, Offset.Y, 48, 48);
    private Rectangle ViewportBounds => new Rectangle(Offset.X, Offset.Y + 70, viewportWidth, viewportHeight);

    public Palette(Vector2 offset)
    {
        Offset = offset;

        // VarsayÄ±lan renkler
        colors.AddRange(new Color[]
        {
            Color.Black, Color.DarkGray, Color.Gray, Color.LightGray, Color.White,
            Color.Red, Color.Maroon, Color.Orange, Color.Gold, Color.Yellow,
            Color.DarkGreen, Color.Green, Color.Lime, Color.DarkBlue, Color.Blue,
            Color.SkyBlue, Color.Purple, Color.Violet, Color.Beige, Color.Brown
        });
    }

    public void AddColor(Color newColor)
    {
        bool exists = colors.Exists(c => c.R == newColor.R && c.G == newColor.G && c.B == newColor.B && c.A == newColor.A);
        if (!exists)
        {
            colors.Add(newColor);
        }
        CurrentColor = newColor;

        float totalHeight = GetTotalContentHeight();
        float maxScroll = Math.Max(0, totalHeight - viewportHeight);
        scrollY = maxScroll;
    }

    public void SetColor(Color newColor)
    {
        CurrentColor = newColor;
    }

    {
        if (index >= 0 && index < colors.Count)
        {
            CurrentColor = colors[index];
            return true;
        }
        return false;

    private float GetTotalContentHeight()
    {
        int totalRows = (int)Math.Ceiling(colors.Count / (float)columns);
        return totalRows * (boxSize + spacing) + 6;
    }

    public bool Update(Vector2 mousePos, Action onOpenColorPicker)
    {
        // 1. Aktif Renk Kutusuna TÄ±klama
        if (Raylib.IsMouseButtonPressed(MouseButton.Left) && Raylib.CheckCollisionPointRec(mousePos, ActiveColorBoxBounds))
        {
            onOpenColorPicker();
            return true;
        }

        Rectangle vp = ViewportBounds;
        bool isInsideViewport = Raylib.CheckCollisionPointRec(mousePos, vp);

        float totalHeight = GetTotalContentHeight();
        float maxScroll = Math.Max(0, totalHeight - viewportHeight);

        // 2. Fare TekerleÄŸi ile KaydÄ±rma
        if (isInsideViewport)
        {
            float wheel = Raylib.GetMouseWheelMove();
            if (wheel != 0)
            {
                scrollY -= wheel * (boxSize + spacing);
                scrollY = Math.Clamp(scrollY, 0, maxScroll);
            }
        }

        // 3. Scrollbar SÃ¼rÃ¼kleme
        if (maxScroll > 0)
        {
            float trackX = vp.X + vp.Width - 8;
            float trackHeight = vp.Height;
            float thumbHeight = Math.Max(24f, (vp.Height / totalHeight) * trackHeight);
            float thumbY = vp.Y + (scrollY / maxScroll) * (trackHeight - thumbHeight);
            Rectangle thumbRect = new Rectangle(trackX, thumbY, 6, thumbHeight);

            if (Raylib.IsMouseButtonPressed(MouseButton.Left) && Raylib.CheckCollisionPointRec(mousePos, thumbRect))
            {
                isDraggingScrollbar = true;
                scrollDragStartMouseY = mousePos.Y;
                scrollDragStartScrollY = scrollY;
            }

            if (isDraggingScrollbar)
            {
                if (Raylib.IsMouseButtonDown(MouseButton.Left))
                {
                    float deltaY = mousePos.Y - scrollDragStartMouseY;
                    float scrollRatio = deltaY / (trackHeight - thumbHeight);
                    scrollY = Math.Clamp(scrollDragStartScrollY + scrollRatio * maxScroll, 0, maxScroll);
                }
                else
                {
                    isDraggingScrollbar = false;
                }
                return true;
            }
        }

        // 4. Renk KutucuklarÄ±na TÄ±klama
        if (isInsideViewport && Raylib.IsMouseButtonPressed(MouseButton.Left))
        {
            float startX = vp.X + 6;
            float startY = vp.Y + 6 - scrollY;

            for (int i = 0; i < colors.Count; i++)
            {
                int col = i % columns;
                int row = i / columns;

                Rectangle boxRect = new Rectangle(
                    startX + col * (boxSize + spacing),
                    startY + row * (boxSize + spacing),
                    boxSize,
                    boxSize
                );

                if (Raylib.CheckCollisionPointRec(mousePos, boxRect))
                {
                    CurrentColor = colors[i];
                    return true;
                }
            }
        }

        return isInsideViewport;
    }

using System.IO;

namespace PixelArtApp.UI;

public static class PixelSkin
{
    public static readonly Color WindowBg      = new Color(30, 30, 34, 255);    
    public static readonly Color PanelBg       = new Color(24, 24, 28, 255);    
    public static readonly Color BorderOutline = new Color(10, 10, 12, 255);    
    public static readonly Color InsetDark     = new Color(18, 18, 22, 255);    

    public static Texture2D CustomButton;
    public static NPatchInfo ButtonPatch;
    public static bool IsCustomLoaded = false;

    public static void LoadSkin()
    {
        string fileName = "Button1.png";
        string exeDir = AppDomain.CurrentDomain.BaseDirectory;
        string fullPath = Path.Combine(exeDir, fileName);

        string targetPath = File.Exists(fileName) ? fileName : (File.Exists(fullPath) ? fullPath : "");

        if (!string.IsNullOrEmpty(targetPath))
        {
            CustomButton = Raylib.LoadTexture(targetPath);
            Raylib.SetTextureFilter(CustomButton, TextureFilter.Point);

            ButtonPatch = new NPatchInfo
            {
                Source = new Rectangle(0, 0, 28, 9),
                Left = 3, Top = 3, Right = 3, Bottom = 3,
                Layout = NPatchLayout.NinePatch
            };
            IsCustomLoaded = true;
        }
    }
    public static void UnloadSkin()
    {
        if (IsCustomLoaded)
        {
            Raylib.UnloadTexture(CustomButton);
            IsCustomLoaded = false;
        }
    }

    public static void DrawButton(Rectangle rect, bool isHover, bool isSelected)
    {
        // YENÄ°: TÄ±klama ve Hover Animasyonu (Fiziksel Ã‡Ã¶kme)
        if (isSelected) 
        {
            rect.Y += 2; // SeÃ§iliyken veya tÄ±klanÄ±rken 2 piksel aÅŸaÄŸÄ± iner
            rect.Height -= 2; 
        }
        else if (isHover)
        {
            rect.Y += 1; // Farenin altÄ±ndayken 1 piksel aÅŸaÄŸÄ± iner (Hafif tepki)
            rect.Height -= 1;
        }

        if (IsCustomLoaded)
        {
            Color tint = isSelected ? new Color(120, 160, 255, 255) : (isHover ? Color.White : new Color(210, 210, 210, 255));
            Raylib.DrawTextureNPatch(CustomButton, ButtonPatch, rect, Vector2.Zero, 0f, tint);
        }
        else
        {
            Color bg = isSelected ? new Color(55, 95, 145, 255) : (isHover ? new Color(60, 60, 68, 255) : new Color(45, 45, 52, 255));
            Raylib.DrawRectangleRec(rect, bg);
            Raylib.DrawRectangleLinesEx(rect, 2, BorderOutline);
            
            if (isSelected) 
                Raylib.DrawRectangle((int)rect.X, (int)rect.Y, (int)rect.Width, 3, new Color(0, 0, 0, 80));
        }
    }
    public static void DrawInsetBox(Rectangle rect)
    {
        Raylib.DrawRectangleRec(rect, InsetDark);
        Raylib.DrawRectangleLinesEx(rect, 2, BorderOutline);
    }

    public static void DrawPanel(Rectangle rect)
    {
        Raylib.DrawRectangleRec(rect, PanelBg);
        Raylib.DrawRectangleLinesEx(rect, 2, BorderOutline);
    }
}

            float curX = 96;
            int tabW = 140;

            for (int i = 0; i < tabs.Count; i++)

                Rectangle closeRect = new Rectangle(tabRect.X + tabW - 26, tabRect.Y + 10, 20, 20);

                if (Raylib.CheckCollisionPointRec(mousePos, closeRect))
                {
                    onCloseTab(i);
                    return true;
                }
                else if (Raylib.CheckCollisionPointRec(mousePos, tabRect))

            if (Raylib.CheckCollisionPointRec(mousePos, plusRect))

                onNewTabClick();
                return true;
            }
        }

        return true;

namespace PixelArtApp.UI;

public class MenuItem
{
    public string Text { get; }
    public Action Action { get; }
    public MenuItem(string text, Action action) { Text = text; Action = action; }
}

public class MenuCategory
{
    public string Title { get; }
    public Rectangle HeaderBounds { get; set; }
    public List<MenuItem> Items { get; } = new();
    public MenuCategory(string title) { Title = title; }
}

public class TopMenu
{
    public float Height { get; } = 36f;
    private readonly List<MenuCategory> categories = new();
    private int openCategoryIndex = -1;

    public void AddCategory(string title, Action<List<MenuItem>> buildItems)
    {
        var category = new MenuCategory(title);
        buildItems(category.Items);
        categories.Add(category);
    }

    public void Layout()
    {
        float currentX = 10f;
        foreach (var cat in categories)
        {
            int textW = Raylib.MeasureText(cat.Title, 15);
            float btnW = textW + 24;
            cat.HeaderBounds = new Rectangle(currentX, 4, btnW, Height - 8);
            currentX += btnW + 5;
        }
    }

    private float CalculateDropdownWidth(MenuCategory cat)
    {
        int maxW = 160; // Minimum geniÅŸlik
        foreach (var item in cat.Items)
        {
            int w = Raylib.MeasureText(item.Text, 14);
            if (w + 40 > maxW) maxW = w + 40; // 40 piksel padding
        }
        return maxW;
    }

    public bool Update(Vector2 mousePos)
    {
        bool mouseOverBar = mousePos.Y < Height;
        bool isAnyMenuOpen = openCategoryIndex != -1;

        if (Raylib.IsMouseButtonPressed(MouseButton.Left))
        {
            for (int i = 0; i < categories.Count; i++)
            {
                if (Raylib.CheckCollisionPointRec(mousePos, categories[i].HeaderBounds))
                {
                    openCategoryIndex = (openCategoryIndex == i) ? -1 : i;
                    return true;
                }
            }

            if (isAnyMenuOpen)
            {
                var cat = categories[openCategoryIndex];
                float dropY = Height;
                float dropW = CalculateDropdownWidth(cat);
                float itemH = 28f;

                for (int j = 0; j < cat.Items.Count; j++)
                {
                    Rectangle itemRect = new Rectangle(cat.HeaderBounds.X, dropY + (j * itemH), dropW, itemH);
                    if (Raylib.CheckCollisionPointRec(mousePos, itemRect))
                    {
                        var action = cat.Items[j].Action;
                        openCategoryIndex = -1;
                        action?.Invoke();
                        return true;
                    }
                }
                openCategoryIndex = -1;
                return true;
            }
        }

        if (isAnyMenuOpen)
        {
            var cat = categories[openCategoryIndex];
            Rectangle dropArea = new Rectangle(cat.HeaderBounds.X, Height, CalculateDropdownWidth(cat), cat.Items.Count * 28f);
            if (Raylib.CheckCollisionPointRec(mousePos, dropArea)) return true;
        }

        return mouseOverBar;
    }

    public void Draw(int screenWidth)
    {
        Vector2 mousePos = Raylib.GetMousePosition();
        Raylib.DrawRectangle(0, 0, screenWidth, (int)Height, new Color(25, 25, 25, 255));
        Raylib.DrawLine(0, (int)Height, screenWidth, (int)Height, Color.DarkGray);

        for (int i = 0; i < categories.Count; i++)
        {
            var cat = categories[i];
            bool isOpen = (openCategoryIndex == i);
            bool isHovered = Raylib.CheckCollisionPointRec(mousePos, cat.HeaderBounds);

            Color bg = isOpen ? new Color(60, 60, 60, 255) : isHovered ? new Color(45, 45, 45, 255) : Color.Blank;
            if (bg.A > 0) Raylib.DrawRectangleRec(cat.HeaderBounds, bg);

            int textW = Raylib.MeasureText(cat.Title, 15);
            Raylib.DrawText(cat.Title, (int)(cat.HeaderBounds.X + (cat.HeaderBounds.Width - textW) / 2), (int)(cat.HeaderBounds.Y + (cat.HeaderBounds.Height - 15) / 2), 15, Color.RayWhite);
        }

        if (openCategoryIndex != -1)
        {
            var cat = categories[openCategoryIndex];
            float dropY = Height;
            float dropW = CalculateDropdownWidth(cat);
            float itemH = 28f;
            float totalH = cat.Items.Count * itemH;

            Raylib.DrawRectangle((int)cat.HeaderBounds.X, (int)dropY, (int)dropW, (int)totalH, new Color(35, 35, 35, 255));
            Raylib.DrawRectangleLines((int)cat.HeaderBounds.X, (int)dropY, (int)dropW, (int)totalH, Color.DarkGray);

            for (int j = 0; j < cat.Items.Count; j++)
            {
                Rectangle itemRect = new Rectangle(cat.HeaderBounds.X, dropY + (j * itemH), dropW, itemH);
                bool itemHovered = Raylib.CheckCollisionPointRec(mousePos, itemRect);
                if (itemHovered) Raylib.DrawRectangleRec(itemRect, new Color(55, 75, 115, 255));
                Raylib.DrawText(cat.Items[j].Text, (int)itemRect.X + 10, (int)itemRect.Y + 6, 14, Color.RayWhite);
            }
        }
    }
}

        {
            items.Add(new MenuItem("Geri Al (Ctrl+Z)", () => { if (ActiveTab != null) { ActiveTab.Selection.CancelOrCommit(ActiveTab.Canvas); ActiveTab.History.Undo(ActiveTab.Canvas); } }));
            items.Add(new MenuItem("Ileri Al (Ctrl+Y)", () => { if (ActiveTab != null) { ActiveTab.Selection.CancelOrCommit(ActiveTab.Canvas); ActiveTab.History.Redo(ActiveTab.Canvas); } }));
            items.Add(new MenuItem("Kopyala (Ctrl+C)", () => ActiveTab?.Selection.Copy(ActiveTab.Canvas)));
            items.Add(new MenuItem("Kes (Ctrl+X)", () => { if (ActiveTab != null) ActiveTab.Selection.Cut(ActiveTab.Canvas, ActiveTab.History); }));
            items.Add(new MenuItem("Yapistir (Ctrl+V)", () => { if (ActiveTab != null) { toolManager.ActiveTool = ToolType.SelectMove; ActiveTab.Selection.Paste(ActiveTab.Canvas, ActiveTab.History); } }));
            items.Add(new MenuItem("Cogalt (Ctrl+D)", () => { if (ActiveTab != null) ActiveTab.Selection.Duplicate(ActiveTab.Canvas, ActiveTab.History); }));
            items.Add(new MenuItem("Temizle", ClearCanvas));
            items.Add(new MenuItem("Ayarlar...", () => settingsDialog.Open()));
        });

        topMenu.AddCategory("Gorunum", items =>
        {
            items.Add(new MenuItem("Izgara Ac/Kapat", ToggleGrid));
            items.Add(new MenuItem("Kamerayi Sifirla", () => { if (ActiveTab != null) CenterCamera(ActiveTab); }));
            items.Add(new MenuItem("Ana Ekran (Home)", () => activeTabIndex = -1));
        });

        topMenu.Layout();

        while (!Raylib.WindowShouldClose() && !shouldExit)
        {
            Update();
            Draw();
        }

        PixelSkin.UnloadSkin();
        Raylib.CloseWindow();
    }

    static void CreateNewTab(int width, int height, Color bgColor, string? customTitle = null)
    {
        string title = customTitle ?? $"Sprite-{untitledCounter++}";
        ProjectTab newTab = new ProjectTab(title, width, height, bgColor, isGridVisible);
        tabs.Add(newTab);
        activeTabIndex = tabs.Count - 1;
        CenterCamera(newTab);
        ShowStatus($"Yeni proje olusturuldu: {width}x{height}");
    }

    static void CloseTab(int index)
    {
        if (index < 0 || index >= tabs.Count) return;
        tabs[index].Selection.CancelOrCommit(tabs[index].Canvas);
        tabs.RemoveAt(index);
        if (tabs.Count == 0) activeTabIndex = -1;
        else if (activeTabIndex >= tabs.Count) activeTabIndex = tabs.Count - 1;
    }

    static void OpenProjectWithDialog()
    {
        string? path = Exporter.OpenImageDialog();
        if (!string.IsNullOrEmpty(path)) LoadProjectFromFile(path);
    }

    static void LoadProjectFromFile(string path)
    {
        if (!File.Exists(path)) return;
        Image img = Raylib.LoadImage(path);
        if (img.Width <= 0 || img.Height <= 0) return;

        int w = Math.Min(img.Width, 256);
        int h = Math.Min(img.Height, 256);
        string fileName = Path.GetFileNameWithoutExtension(path);
        ProjectTab newTab = new ProjectTab(fileName, w, h, Canvas.TransparentColor, isGridVisible) { FilePath = path };

        for (int x = 0; x < w; x++)
        {
            for (int y = 0; y < h; y++)
            {
                Color c = Raylib.GetImageColor(img, x, y);
                newTab.Canvas.SetPixel(x, y, c);
            }
        }
        Raylib.UnloadImage(img);
        newTab.History.SaveState(newTab.Canvas);

        tabs.Add(newTab);
        activeTabIndex = tabs.Count - 1;
        CenterCamera(newTab);
        homeScreen.AddRecentFile(path);
        ShowStatus($"Acildi: {Path.GetFileName(path)}");
    }

    static void SelectTool(ToolType newTool)
    {
        if (ActiveTab != null && toolManager.ActiveTool == ToolType.SelectMove && newTool != ToolType.SelectMove)
            ActiveTab.Selection.CancelOrCommit(ActiveTab.Canvas);
        toolManager.ActiveTool = newTool;
        sizeInputActive = false;
    }

    static void ToggleGrid()
    {
        isGridVisible = !isGridVisible;
        foreach (var t in tabs) t.Canvas.ShowGrid = isGridVisible;
    }

    static void ClearCanvas()
    {
        if (ActiveTab == null) return;
        ActiveTab.Selection.CancelOrCommit(ActiveTab.Canvas);
        ActiveTab.History.SaveState(ActiveTab.Canvas);
        ActiveTab.Canvas.Clear(Canvas.TransparentColor);
        ShowStatus("Tuval temizlendi.");
    }

    static void ShowStatus(string msg) { statusMessage = msg; messageTimer = 3.0f; }

    static void CenterCamera(ProjectTab tab)
    {
        int sw = Raylib.GetScreenWidth();
        int sh = Raylib.GetScreenHeight();
        float centerAreaX = (toolPanel.Width + (sw - RightPanelWidth)) / 2f;
        float centerAreaY = (WorkspaceTopY + animationPanel.Height + (sh - layerPanel.Height)) / 2f;
        Camera2D cam = new Camera2D();
        cam.Target = new Vector2((tab.Canvas.Width * tab.Canvas.PixelScale) / 2f, (tab.Canvas.Height * tab.Canvas.PixelScale) / 2f);
        cam.Offset = new Vector2(centerAreaX, centerAreaY);
        cam.Rotation = 0.0f;
        cam.Zoom = ZoomLevels[tab.ZoomIndex];
        targetZoom = cam.Zoom;
        tab.Camera = cam;
    }

    static void ApplyTypedSize()
    {
        if (int.TryParse(sizeInputText, out int val))
        {
            val = Math.Clamp(val, 1, 50);
            if (toolManager.ActiveTool == ToolType.Eraser) toolManager.EraserSize = val;
            else toolManager.PenSize = val;
            ShowStatus($"Boyut ayarlandi: {val}px");
        }
        sizeInputActive = false;
    }

    static void Update()
    {
        int sw = Raylib.GetScreenWidth();
        int sh = Raylib.GetScreenHeight();
        Vector2 mouseScreenPos = Raylib.GetMousePosition();

        if (Raylib.IsKeyPressed(KeyboardKey.F11)) Raylib.ToggleFullscreen();

        palette.Offset = new Vector2(sw - RightPanelWidth + 20, WorkspaceTopY + 35);

        if (newProjectDialog.IsOpen)
        {
            newProjectDialog.Update(mouseScreenPos, (w, h, bg) => CreateNewTab(w, h, bg));
            return;
        }

        if (exportDialog.IsOpen)
        {
            exportDialog.Update(mouseScreenPos, (path, scale, allFrames, vertical) =>
            {
                var tab = ActiveTab;
                if (tab != null)
                {
                    bool success = Exporter.SaveExport(tab.Canvas, path, scale, allFrames, vertical);
                    if (success)
                    {
                        tab.FilePath = path;
                        tab.Title = Path.GetFileNameWithoutExtension(path);
                        homeScreen.AddRecentFile(path);
                        ShowStatus($"Basariyla aktarildi: {Path.GetFileName(path)}");
                    }
                    else ShowStatus("Disa aktarma basarisiz oldu veya iptal edildi.");
                }
            });
            return;
        }

        if (settingsDialog.IsOpen)
        {
            settingsDialog.Update(mouseScreenPos, ToggleGrid);
            return;
        }

        if (colorPicker.IsOpen)
        {
            colorPicker.Update(mouseScreenPos, (newColor, addToPalette) =>
            {
                if (addToPalette) { palette.AddColor(newColor); ShowStatus("Ozel renk palete eklendi!"); }
                else { palette.SetColor(newColor); ShowStatus("Yeni renk secildi."); }
            });
            return;
        }

        bool isCtrlDown = Raylib.IsKeyDown(KeyboardKey.LeftControl) || Raylib.IsKeyDown(KeyboardKey.RightControl);

        if (isCtrlDown && Raylib.IsKeyPressed(KeyboardKey.N)) { newProjectDialog.Open(); return; }
        if (isCtrlDown && Raylib.IsKeyPressed(KeyboardKey.O)) { OpenProjectWithDialog(); return; }
        if (isCtrlDown && Raylib.IsKeyPressed(KeyboardKey.W) && activeTabIndex >= 0) { CloseTab(activeTabIndex); return; }

        bool topMenuIntercepted = topMenu.Update(mouseScreenPos);

        bool tabBarIntercepted = false;
        if (!topMenuIntercepted)
        {
            tabBarIntercepted = tabBar.Update(
                mouseScreenPos, tabs, activeTabIndex,
                newIndex => { ActiveTab?.Selection.CancelOrCommit(ActiveTab.Canvas); activeTabIndex = newIndex; if (ActiveTab != null) targetZoom = ZoomLevels[ActiveTab.ZoomIndex]; },
                CloseTab, () => newProjectDialog.Open()
            );
        }

        if (ActiveTab == null)
        {
            if (!topMenuIntercepted && !tabBarIntercepted)
                homeScreen.Update(mouseScreenPos, sw, sh, WorkspaceTopY, () => newProjectDialog.Open(), OpenProjectWithDialog, LoadProjectFromFile);
            return;
        }

        Canvas canvas = ActiveTab.Canvas;
        SelectionManager selection = ActiveTab.Selection;
        HistoryManager history = ActiveTab.History;
        Camera2D camera = ActiveTab.Camera;

        // --- YENİ: Yumuşak (Smooth) Zoom Animasyonu ---
        targetZoom = ZoomLevels[ActiveTab.ZoomIndex];
        if (Math.Abs(camera.Zoom - targetZoom) > 0.005f)
        {
            camera.Zoom = Raymath.Lerp(camera.Zoom, targetZoom, Raylib.GetFrameTime() * 14f);
            ActiveTab.Camera = camera;
        }
        else if (camera.Zoom != targetZoom)
        {
            camera.Zoom = targetZoom;
            ActiveTab.Camera = camera;
        }

        // --- YENİ: Sağ Panelde Elle Boyut Yazma Kutusu Etkileşimi ---
        int rightPanelX = sw - RightPanelWidth;
        Rectangle sizeBoxRect = new Rectangle(rightPanelX + 75, 518, 45, 22);
        Rectangle btnMinusRect = new Rectangle(rightPanelX + 52, 518, 20, 22);
        Rectangle btnPlusRect = new Rectangle(rightPanelX + 123, 518, 20, 22);

        if (Raylib.IsMouseButtonPressed(MouseButton.Left))
        {
            if (Raylib.CheckCollisionPointRec(mouseScreenPos, sizeBoxRect))
            {
                sizeInputActive = true;
                int curSize = toolManager.ActiveTool == ToolType.Eraser ? toolManager.EraserSize : toolManager.PenSize;
                sizeInputText = curSize.ToString();
                return;
            }
            else if (Raylib.CheckCollisionPointRec(mouseScreenPos, btnMinusRect))
            {
                if (toolManager.ActiveTool == ToolType.Eraser)
                    toolManager.EraserSize = Math.Clamp(toolManager.EraserSize - 1, 1, 50);
                else
                    toolManager.PenSize = Math.Clamp(toolManager.PenSize - 1, 1, 50);
                sizeInputActive = false;
                return;
            }
            else if (Raylib.CheckCollisionPointRec(mouseScreenPos, btnPlusRect))
            {
                if (toolManager.ActiveTool == ToolType.Eraser)
                    toolManager.EraserSize = Math.Clamp(toolManager.EraserSize + 1, 1, 50);
                else
                    toolManager.PenSize = Math.Clamp(toolManager.PenSize + 1, 1, 50);
                sizeInputActive = false;
                return;
            }
            else if (sizeInputActive)
            {
                ApplyTypedSize();
            }
        }

        if (sizeInputActive)
        {
            int key = Raylib.GetCharPressed();
            while (key > 0)
            {
                if (key >= '0' && key <= '9' && sizeInputText.Length < 2)
                {
                    sizeInputText += (char)key;
                }
                key = Raylib.GetCharPressed();
            }
            if (Raylib.IsKeyPressed(KeyboardKey.Backspace) && sizeInputText.Length > 0)
            {
                sizeInputText = sizeInputText[..^1];
            }
            if (Raylib.IsKeyPressed(KeyboardKey.Enter) || Raylib.IsKeyPressed(KeyboardKey.KpEnter) || Raylib.IsKeyPressed(KeyboardKey.Escape))
            {
                ApplyTypedSize();
            }
            return; // Yazı yazarken kısayolların tetiklenmesini engelle
        }

        if (Raylib.IsKeyPressed(AppSettings.KeyToggleLayers)) layerPanel.ToggleHidden();

        if (selection.HasSelection && !selection.IsCreatingSelection && !selection.IsDraggingFloating)
        {
            Rectangle curBounds = selection.GetCurrentBounds();
            Vector2 bottomCenterWorld = new Vector2(
                canvas.Offset.X + (curBounds.X + curBounds.Width / 2f) * canvas.PixelScale,
                canvas.Offset.Y + (curBounds.Y + curBounds.Height) * canvas.PixelScale
            );
            Vector2 screenAnchor = Raylib.GetWorldToScreen2D(bottomCenterWorld, camera);
            Rectangle menuRect = new Rectangle(Math.Clamp(screenAnchor.X - 142.5f, 150, sw - 190 - 285), Math.Clamp(screenAnchor.Y + 12, 45, sh - 45), 285, 32);

            if (Raylib.CheckCollisionPointRec(mouseScreenPos, menuRect))
                if (Raylib.IsMouseButtonPressed(MouseButton.Left) || Raylib.IsMouseButtonDown(MouseButton.Left)) return;
        }

        bool animPanelIntercepted = false;
        if (!topMenuIntercepted && !tabBarIntercepted)
           animPanelIntercepted = animationPanel.Update(canvas, history, mouseScreenPos, sw, sh, () => { selection.CancelOrCommit(canvas); ShowStatus($"Aktif Kare: {canvas.ActiveFrameIndex + 1}"); });

        bool toolPanelIntercepted = toolPanel.Update(
            mouseScreenPos, SelectTool, shape => toolManager.ActiveShape = shape,
            selMode => { selection.CancelOrCommit(canvas); toolManager.ActiveSelectionMode = selMode; }
        );

        bool paletteIntercepted = palette.Update(mouseScreenPos, () => colorPicker.Open(palette.CurrentColor));
        bool layerPanelIntercepted = layerPanel.Update(canvas, history, mouseScreenPos, sw, sh, () => { selection.CancelOrCommit(canvas); ShowStatus($"Aktif Katman: {canvas.Layers[canvas.ActiveLayerIndex].Name}"); });

        bool insideDrawingArea = !topMenuIntercepted && !tabBarIntercepted && !animPanelIntercepted &&
                                 !toolPanelIntercepted && !paletteIntercepted && !layerPanelIntercepted &&
                                 mouseScreenPos.X >= toolPanel.Width && mouseScreenPos.X < (sw - RightPanelWidth) &&
                                 mouseScreenPos.Y >= WorkspaceTopY;

        float wheel = Raylib.GetMouseWheelMove();
        if (wheel != 0 && insideDrawingArea)
        {
            if (isCtrlDown)
            {
                if (toolManager.ActiveTool == ToolType.Eraser)
                {
                    toolManager.EraserSize = Math.Clamp(toolManager.EraserSize + (int)Math.Sign(wheel), 1, 50);
                    ShowStatus($"Silgi Boyutu: {toolManager.EraserSize}px");
                }
                else
                {
                    toolManager.PenSize = Math.Clamp(toolManager.PenSize + (int)Math.Sign(wheel), 1, 50);
                    ShowStatus($"Kalem Boyutu: {toolManager.PenSize}px");
                }
            }
            else
            {
                int prevIndex = ActiveTab.ZoomIndex;
                if (wheel > 0 && ActiveTab.ZoomIndex < ZoomLevels.Length - 1) ActiveTab.ZoomIndex++;
                else if (wheel < 0 && ActiveTab.ZoomIndex > 0) ActiveTab.ZoomIndex--;

                if (prevIndex != ActiveTab.ZoomIndex)
                {
                    camera.Offset = mouseScreenPos;
                    camera.Target = Raylib.GetScreenToWorld2D(mouseScreenPos, camera);
                    targetZoom = ZoomLevels[ActiveTab.ZoomIndex];
                    ActiveTab.Camera = camera;
                }
            }
        }

        if (Raylib.IsMouseButtonDown(MouseButton.Middle))
        {
            Vector2 delta = Raylib.GetMouseDelta();
            delta = Vector2.Multiply(delta, -1.0f / camera.Zoom);
            camera.Target = Vector2.Add(camera.Target, delta);
            ActiveTab.Camera = camera;
        }

        if (!isCtrlDown && !topMenuIntercepted && !tabBarIntercepted && !paletteIntercepted && !toolPanelIntercepted)
        {
            for (int i = 0; i < AppSettings.KeyColors.Length; i++)
            {
                if (AppSettings.KeyColors[i] != KeyboardKey.Null && Raylib.IsKeyPressed(AppSettings.KeyColors[i]))
                {
                    if (palette.SelectColor(i)) ShowStatus($"Renk {i + 1} secildi.");
                }
            }

            if (Raylib.IsKeyPressed(AppSettings.KeyPen)) SelectTool(ToolType.Pen);
            if (Raylib.IsKeyPressed(AppSettings.KeyEraser)) SelectTool(ToolType.Eraser);
            if (Raylib.IsKeyPressed(AppSettings.KeyBucket)) SelectTool(ToolType.Bucket);
            if (Raylib.IsKeyPressed(AppSettings.KeySmartBucket)) SelectTool(ToolType.SmartBucket);
            if (Raylib.IsKeyPressed(AppSettings.KeyEyedropper)) SelectTool(ToolType.Eyedropper);
            if (Raylib.IsKeyPressed(AppSettings.KeyLine)) SelectTool(ToolType.Line);
            if (Raylib.IsKeyPressed(AppSettings.KeyShape)) SelectTool(ToolType.Shape);
            if (Raylib.IsKeyPressed(AppSettings.KeySelectMove)) SelectTool(ToolType.SelectMove);
        }

        if (isCtrlDown && Raylib.IsKeyPressed(KeyboardKey.Z)) { selection.CancelOrCommit(canvas); history.Undo(canvas); }
        else if (isCtrlDown && Raylib.IsKeyPressed(KeyboardKey.Y)) { selection.CancelOrCommit(canvas); history.Redo(canvas); }
        else if (isCtrlDown && Raylib.IsKeyPressed(KeyboardKey.S)) { if (ActiveTab != null) exportDialog.Open(ActiveTab.Title, ActiveTab.Canvas); }
        else if (isCtrlDown && Raylib.IsKeyPressed(KeyboardKey.C) && selection.HasSelection) { selection.Copy(canvas); ShowStatus("Secim kopyalandi"); }
        else if (isCtrlDown && Raylib.IsKeyPressed(KeyboardKey.X) && selection.HasSelection) { selection.Cut(canvas, history); ShowStatus("Secim kesildi"); }
        else if (isCtrlDown && Raylib.IsKeyPressed(KeyboardKey.V)) { SelectTool(ToolType.SelectMove); selection.Paste(canvas, history); ShowStatus("Yapistirildi"); }
        else if (isCtrlDown && Raylib.IsKeyPressed(KeyboardKey.D) && selection.HasSelection) { selection.Duplicate(canvas, history); ShowStatus("Secim cogaltildi"); }

        if ((Raylib.IsKeyPressed(KeyboardKey.Delete) || Raylib.IsKeyPressed(KeyboardKey.Backspace)) && selection.HasSelection) { selection.DeleteSelection(canvas, history); ShowStatus("Secim silindi"); }
        if ((Raylib.IsKeyPressed(KeyboardKey.Escape) || Raylib.IsKeyPressed(KeyboardKey.Enter)) && selection.HasSelection) { selection.CancelOrCommit(canvas); ShowStatus("Secim kapatildi"); }

        Vector2 worldMousePos = Raylib.GetScreenToWorld2D(mouseScreenPos, camera);
        var (gx, gy) = canvas.ScreenToGrid(worldMousePos);

        if (!Raylib.IsMouseButtonDown(MouseButton.Left) && !Raylib.IsMouseButtonDown(MouseButton.Right)) lastGridPos = null;

        if (toolManager.ActiveTool == ToolType.SelectMove)
        {
            bool isShift = Raylib.IsKeyDown(KeyboardKey.LeftShift) || Raylib.IsKeyDown(KeyboardKey.RightShift);
            bool isAlt = Raylib.IsKeyDown(KeyboardKey.LeftAlt) || Raylib.IsKeyDown(KeyboardKey.RightAlt);
            bool isAdd = isShift;
            bool isSubtract = isCtrlDown || isAlt;

            if (insideDrawingArea && Raylib.IsMouseButtonPressed(MouseButton.Left))
            {
                if (!isAdd && !isSubtract && selection.HasSelection && selection.IsPointInCurrentSelection(gx, gy)) selection.StartMove(canvas, history, gx, gy);
                else if (canvas.IsMouseInside(worldMousePos)) selection.StartSelection(canvas, gx, gy, toolManager.ActiveSelectionMode, isAdd, isSubtract);
                else selection.CancelOrCommit(canvas);
            }

            if (Raylib.IsMouseButtonDown(MouseButton.Left))
            {
                if (selection.IsDraggingFloating) selection.UpdateMove(gx, gy);
                else if (selection.IsCreatingSelection) selection.UpdateSelectionCreation(gx, gy, toolManager.ActiveSelectionMode);
            }
            if (Raylib.IsMouseButtonReleased(MouseButton.Left))
            {
                if (selection.IsDraggingFloating) selection.EndMove();
                else if (selection.IsCreatingSelection) selection.EndSelectionCreation(canvas, toolManager.ActiveSelectionMode, isAdd, isSubtract);
            }
        }
        else if (insideDrawingArea && canvas.IsMouseInside(worldMousePos))
        {
            if (toolManager.ActiveTool == ToolType.Eyedropper || Raylib.IsKeyDown(KeyboardKey.LeftAlt))
            {
                if (Raylib.IsMouseButtonPressed(MouseButton.Left))
                {
                    Color pickedColor = canvas.GetMergedPixel(gx, gy);
                    if (pickedColor.A > 0) { palette.SetColor(pickedColor); ShowStatus($"Renk secildi: R:{pickedColor.R}"); }
                    if (toolManager.ActiveTool == ToolType.Eyedropper) toolManager.ActiveTool = ToolType.Pen;
                }
            }
            else if (toolManager.ActiveTool == ToolType.Bucket || toolManager.ActiveTool == ToolType.SmartBucket)
            {
                if (Raylib.IsMouseButtonPressed(MouseButton.Left)) { history.SaveState(canvas); toolManager.ApplyTool(canvas, gx, gy, palette.CurrentColor); }
            }
            else if (toolManager.ActiveTool == ToolType.Line || toolManager.ActiveTool == ToolType.Shape)
            {
                if (Raylib.IsMouseButtonPressed(MouseButton.Left)) dragStartPos = (gx, gy);
                else if (Raylib.IsMouseButtonReleased(MouseButton.Left) && dragStartPos.HasValue)
                {
                    history.SaveState(canvas);
                    List<(int x, int y)> shapePixels = (toolManager.ActiveTool == ToolType.Line)
                        ? ToolManager.GetLinePixels(dragStartPos.Value.x, dragStartPos.Value.y, gx, gy)
                        : ToolManager.GetShapePixels(dragStartPos.Value.x, dragStartPos.Value.y, gx, gy, toolManager.ActiveShape);
                    foreach (var (px, py) in shapePixels) canvas.SetPixel(px, py, palette.CurrentColor);
                    dragStartPos = null;
                }
            }
            else
            {
                if (Raylib.IsMouseButtonPressed(MouseButton.Left) || Raylib.IsMouseButtonPressed(MouseButton.Right)) history.SaveState(canvas);
                if (Raylib.IsMouseButtonDown(MouseButton.Left))
                {
                    if (lastGridPos.HasValue) toolManager.DrawLine(canvas, lastGridPos.Value.x, lastGridPos.Value.y, gx, gy, palette.CurrentColor);
                    else toolManager.ApplyTool(canvas, gx, gy, palette.CurrentColor);
                    lastGridPos = (gx, gy);
                }
                else if (Raylib.IsMouseButtonDown(MouseButton.Right))
                {
                    if (lastGridPos.HasValue) toolManager.DrawLine(canvas, lastGridPos.Value.x, lastGridPos.Value.y, gx, gy, Canvas.TransparentColor);
                    else canvas.SetPixel(gx, gy, Canvas.TransparentColor);
                    lastGridPos = (gx, gy);
                }
            }
        }
        else if (Raylib.IsMouseButtonReleased(MouseButton.Left)) dragStartPos = null;

        if (messageTimer > 0)
        {
            messageTimer -= Raylib.GetFrameTime();
            if (messageTimer <= 0) statusMessage = "Hazir";
        }
    }

    static void Draw()
    {
        int sw = Raylib.GetScreenWidth();
        int sh = Raylib.GetScreenHeight();
        Vector2 mouseScreenPos = Raylib.GetMousePosition();

        Raylib.BeginDrawing();
        Raylib.ClearBackground(PixelSkin.WindowBg);

        if (ActiveTab == null) homeScreen.Draw(sw, sh, WorkspaceTopY);
        else
        {
            Canvas canvas = ActiveTab.Canvas;
            SelectionManager selection = ActiveTab.Selection;
            HistoryManager history = ActiveTab.History;
            Camera2D camera = ActiveTab.Camera;

            Raylib.BeginMode2D(camera);
            canvas.Draw();

            Vector2 worldMousePos = Raylib.GetScreenToWorld2D(mouseScreenPos, camera);
            var (gx, gy) = canvas.ScreenToGrid(worldMousePos);

            // Çizgi ve Şekil Önizlemesi
            if (dragStartPos.HasValue && (toolManager.ActiveTool == ToolType.Line || toolManager.ActiveTool == ToolType.Shape))
            {
                List<(int x, int y)> previewPixels = (toolManager.ActiveTool == ToolType.Line)
                    ? ToolManager.GetLinePixels(dragStartPos.Value.x, dragStartPos.Value.y, gx, gy)
                    : ToolManager.GetShapePixels(dragStartPos.Value.x, dragStartPos.Value.y, gx, gy, toolManager.ActiveShape);
                Color previewColor = new Color(palette.CurrentColor.R, palette.CurrentColor.G, palette.CurrentColor.B, (byte)180);
                foreach (var (px, py) in previewPixels)
                    if (px >= 0 && px < canvas.Width && py >= 0 && py < canvas.Height)
                        Raylib.DrawRectangle((int)canvas.Offset.X + px * canvas.PixelScale, (int)canvas.Offset.Y + py * canvas.PixelScale, canvas.PixelScale, canvas.PixelScale, previewColor);
            }

            // --- YENİ: Boyanacak Yeri Gösteren İçi Boş Kare (Fırça Önizlemesi + Nefes Alma Animasyonu) ---
            if (canvas.IsMouseInside(worldMousePos) && (toolManager.ActiveTool == ToolType.Pen || toolManager.ActiveTool == ToolType.Eraser))
            {
                int size = toolManager.ActiveTool == ToolType.Pen ? toolManager.PenSize : toolManager.EraserSize;
                int offset = size / 2;
                int startX = gx - offset;
                int startY = gy - offset;

                Rectangle brushPreviewRect = new Rectangle(
                    canvas.Offset.X + startX * canvas.PixelScale,
                    canvas.Offset.Y + startY * canvas.PixelScale,
                    size * canvas.PixelScale,
                    size * canvas.PixelScale
                );

                // Nefes alma (Pulse) animasyonu
                float pulse = (float)(Math.Sin(Raylib.GetTime() * 8.0) * 0.5 + 0.5);
                byte alphaPulse = (byte)(150 + pulse * 105);

                // Kalemdeyken hafif şeffaf renk önizlemesi
                if (toolManager.ActiveTool == ToolType.Pen)
                {
                    Color ghostColor = new Color(palette.CurrentColor.R, palette.CurrentColor.G, palette.CurrentColor.B, (byte)85);
                    Raylib.DrawRectangleRec(brushPreviewRect, ghostColor);
                }

                // Dış siyah çerçeve ve iç parlayan çerçeve (her zeminde net görünmesi için)
                float lineThickness = Math.Max(1f, 1.5f / camera.Zoom);
                Color outlineCol = toolManager.ActiveTool == ToolType.Eraser 
                    ? new Color((byte)255, (byte)100, (byte)100, alphaPulse) 
                    : new Color((byte)255, (byte)255, (byte)255, alphaPulse);

                Raylib.DrawRectangleLinesEx(brushPreviewRect, lineThickness * 1.6f, new Color((byte)0, (byte)0, (byte)0, (byte)200));
                Raylib.DrawRectangleLinesEx(brushPreviewRect, lineThickness, outlineCol);
            }

            selection.DrawWorldOverlay(canvas);
            Raylib.EndMode2D();
            selection.CheckAndDrawFloatingMenu(canvas, camera, mouseScreenPos, history, ShowStatus);

            toolPanel.Draw(toolManager.ActiveTool, toolManager.ActiveShape, toolManager.ActiveSelectionMode, sh);

            // Animasyon ve Katman Panellerini Çiz
            animationPanel.Draw(canvas, sw, sh);
            layerPanel.Draw(canvas, sw, sh);

            int rightPanelX = sw - RightPanelWidth;
            Rectangle rightPanelRect = new Rectangle(rightPanelX, WorkspaceTopY, RightPanelWidth, sh - WorkspaceTopY);
            PixelSkin.DrawPanel(rightPanelRect);

            palette.Draw();

            // Sağ Alt Bilgi ve Boyut Ayar Kutusu
            Rectangle infoBox = new Rectangle(rightPanelX + 10, 430, RightPanelWidth - 20, 145);
            PixelSkin.DrawInsetBox(infoBox);

            Raylib.DrawText($"Tuval: {canvas.Width}x{canvas.Height}", rightPanelX + 20, 442, 13, Color.LightGray);
            string toolDesc = toolManager.ActiveTool switch
            {
                ToolType.Shape => $"Sekil ({toolManager.ActiveShape})",
                ToolType.SelectMove => $"Sec ({toolManager.ActiveSelectionMode})",
                _ => toolManager.ActiveTool.ToString()
            };
            Raylib.DrawText($"Arac: {toolDesc}", rightPanelX + 20, 466, 12, Color.Yellow);
            Raylib.DrawText($"Zoom: {ZoomLevels[ActiveTab.ZoomIndex]}x", rightPanelX + 20, 490, 13, Color.LightGray);

            // --- YENİ: Elle Fırça/Silgi Boyutu Yazma Alanı ---
            Raylib.DrawText("Boyut:", rightPanelX + 16, 523, 12, Color.LightGray);

            Rectangle btnMinus = new Rectangle(rightPanelX + 52, 518, 20, 22);
            Rectangle sizeBox = new Rectangle(rightPanelX + 75, 518, 45, 22);
            Rectangle btnPlus = new Rectangle(rightPanelX + 123, 518, 20, 22);

            PixelSkin.DrawButton(btnMinus, Raylib.CheckCollisionPointRec(mouseScreenPos, btnMinus), false);
            Raylib.DrawText("-", (int)btnMinus.X + 7, (int)btnMinus.Y + 4, 14, Color.White);

            PixelSkin.DrawInsetBox(sizeBox);
            if (sizeInputActive)
            {
                Raylib.DrawRectangleLinesEx(sizeBox, 1, Color.Yellow);
                string blink = ((int)(Raylib.GetTime() * 3) % 2 == 0) ? "|" : "";
                Raylib.DrawText(sizeInputText + blink, (int)sizeBox.X + 8, (int)sizeBox.Y + 5, 12, Color.Yellow);
            }
            else
            {
                int currentToolSize = toolManager.ActiveTool == ToolType.Eraser ? toolManager.EraserSize : toolManager.PenSize;
                Raylib.DrawText($"{currentToolSize}px", (int)sizeBox.X + 8, (int)sizeBox.Y + 5, 12, Color.RayWhite);
            }

            PixelSkin.DrawButton(btnPlus, Raylib.CheckCollisionPointRec(mouseScreenPos, btnPlus), false);
            Raylib.DrawText("+", (int)btnPlus.X + 6, (int)btnPlus.Y + 4, 14, Color.White);

            if (statusMessage != "Hazir") Raylib.DrawText(statusMessage, rightPanelX + 20, 550, 11, Color.Green);
        }

        tabBar.Draw(tabs, activeTabIndex, sw);
        topMenu.Draw(sw);

        colorPicker.Draw();
        newProjectDialog.Draw();
        settingsDialog.Draw(isGridVisible);
        exportDialog.Draw();

        Raylib.EndDrawing();
    }
